18template <typename Dim>
19inline Dim resolveAnchor(Anchor anchor, Dim first, Dim last) {
20 switch (anchor) {
21 case ANCHOR_MIN:
22 return first;
23 case ANCHOR_MID:
24 return (first + last) / 2;
25 case ANCHOR_MAX:
26 return last;
27 case ANCHOR_ORIGIN:
28 default:
29 return 0;
30 }
31}

33class AlignBase {
34 public:
35 explicit constexpr AlignBase() : AlignBase(ANCHOR_ORIGIN, ANCHOR_ORIGIN, 0) {}
36
37 explicit constexpr AlignBase(Anchor dst, Anchor src, int16_t shift)
38 : rep_((uint16_t)(shift + (1 << 11)) << 4 | src << 2 | dst) {}
39
40 constexpr Anchor src() const { return (Anchor)((rep_ >> 2) & 3); }
41 constexpr Anchor dst() const { return (Anchor)(rep_ & 3); }
42
43 constexpr int16_t shift() const { return (int16_t)((rep_ >> 4) - (1 << 11)); }

47 Dim last_inner) const {

66// Represents a horizontal alignment. Consists of three components:
67// 1. the 'source' anchor, determining which coordinate of the source should get
68// aligned (left, center, right, or the origin)
69// 2. the 'destination' anchor, determining to which coordinate of the
70// destination should the source be aligned (like above)
71// 3. an additional absolute offset (padding).
72//
73// You generally shouldn't construct these directly. Instead, start with one
74// of the constants and customize it, e.g.:
75//
76// kLeft.toCenter().shiftBy(5) - aligns the left boundary of the source to be
77// 5 pixels to the right of the center of the destination.
78class HAlign : public internal::AlignBase {
79 public:
80 using AlignBase::AlignBase;
81
82 // Applies an extra horizontal shift.
83 constexpr HAlign shiftBy(int16_t shift_by) const {
84 return HAlign(dst(), src(), shift() + shift_by);
85 }
86
87 // Sets the anchor relative to the left of the destination.
88 constexpr HAlign toLeft() const { return HAlign(ANCHOR_MIN, src(), shift()); }
89
90 // Sets the anchor relative to the center of the destination.
91 constexpr HAlign toCenter() const {
92 return HAlign(ANCHOR_MID, src(), shift());
93 }
94
95 // Sets the anchor relative to the right of the destination.
96 constexpr HAlign toRight() const {
97 return HAlign(ANCHOR_MAX, src(), shift());
98 }
99
100 // Sets the anchor relative to the origin (point zero) of the destination.
101 constexpr HAlign toOrigin() const {
102 return HAlign(ANCHOR_ORIGIN, src(), shift());
103 }
104};
