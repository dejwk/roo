122void BackgroundFillOptimizer::writeRects(BlendingMode mode, Color* color,
123 int16_t* x0, int16_t* y0, int16_t* x1,
124 int16_t* y1, uint16_t count) {
125 BufferedRectWriter writer(output_, mode);
126 while (count-- > 0) {
127 Color c = *color++;
128 uint8_t palette_idx = getIdxInPalette(c, palette_, palette_size_);
129 if (palette_idx != 0) {
130 RectFillWriter adapter{
131 .color = c,
132 .writer = writer,
133 };
134 fillRectBg(*x0++, *y0++, *x1++, *y1++, &adapter, palette_idx);
135 } else {
136 // Not a background palette color -> clear the nibble subrectangle
137 // corresponding to a region entirely enclosing the the drawn rectangle
138 // before writing to the underlying device.
139 background_mask_->fillRect(Box(*x0 / kBgFillOptimizerWindowSize,
140 *y0 / kBgFillOptimizerWindowSize,
141 *x1 / kBgFillOptimizerWindowSize,
142 *y1 / kBgFillOptimizerWindowSize),
143 0);
144 writer.writeRect(*x0++, *y0++, *x1++, *y1++, c);
145 }
146 }

173void BackgroundFillOptimizer::writePixels(BlendingMode mode, Color* color,
174 int16_t* x, int16_t* y,
175 uint16_t pixel_count) {
176 int16_t* x_out = x;
177 int16_t* y_out = y;
178 Color* color_out = color;
179 uint16_t new_pixel_count = 0;
180 for (uint16_t i = 0; i < pixel_count; ++i) {
181 uint8_t palette_idx = getIdxInPalette(color[i], palette_, palette_size_);
182 if (palette_idx != 0) {
183 // Do not actually draw the background pixel if the corresponding
184 // bit-mask rectangle is known to be all-background already.
185 if (background_mask_->get(x[i] / kBgFillOptimizerWindowSize,
186 y[i] / kBgFillOptimizerWindowSize) ==
187 palette_idx)
188 continue;
189 }
190 // Mark the corresponding bit-mask rectangle as no longer
191 // all-background.
192 background_mask_->set(x[i] / kBgFillOptimizerWindowSize,
193 y[i] / kBgFillOptimizerWindowSize, 0);
194 *x_out++ = x[i];
195 *y_out++ = y[i];
196 *color_out++ = color[i];
197 new_pixel_count++;
198 }
199 if (new_pixel_count > 0) {
200 output_.writePixels(mode, color, x, y, new_pixel_count);
201 }
202}

206 uint16_t pixel_count) {

208 if (palette_idx != 0) {
209 int16_t* x_out = x;
210 int16_t* y_out = y;
211 uint16_t new_pixel_count = 0;
212 for (uint16_t i = 0; i < pixel_count; ++i) {
213 // Do not actually draw the background pixel if the corresponding

215 if (background_mask_->get(x[i] / kBgFillOptimizerWindowSize,
216 y[i] / kBgFillOptimizerWindowSize) ==
217 palette_idx) {
218 continue;

220 background_mask_->set(x[i] / kBgFillOptimizerWindowSize,
221 y[i] / kBgFillOptimizerWindowSize, 0);
222 *x_out++ = x[i];

258template <typename Filler>
259void BackgroundFillOptimizer::fillRectBg(int16_t x0, int16_t y0, int16_t x1,
260 int16_t y1, Filler* filler,
261 uint8_t palette_idx) {
262 // Iterate over the bit-map rectangle that encloses the requested rectangle.
263 // Skip writing sub-rectangles that are known to be already all-background.
264 Box filter_box(
265 x0 / kBgFillOptimizerWindowSize, y0 / kBgFillOptimizerWindowSize,
266 x1 / kBgFillOptimizerWindowSize, y1 / kBgFillOptimizerWindowSize);
267 internal::NibbleRectWindowIterator itr(background_mask_, filter_box.xMin(),
268 filter_box.yMin(), filter_box.xMax(),
269 filter_box.yMax());
270 int16_t xendl = filter_box.xMax() + 1;
271 for (int16_t y = filter_box.yMin(); y <= filter_box.yMax(); ++y) {
272 int16_t xstart = -1;
273 for (int16_t x = filter_box.xMin(); x <= xendl; ++x) {
274 if (x == xendl || itr.next() == palette_idx) {
275 // End of the streak. Let's draw the necessary rectangle.
276 if (xstart >= 0) {
277 Box box(
278 xstart * kBgFillOptimizerWindowSize,
279 y * kBgFillOptimizerWindowSize,
280 x * kBgFillOptimizerWindowSize - 1,
281 y * kBgFillOptimizerWindowSize + kBgFillOptimizerWindowSize - 1);
282 if (box.clip(Box(x0, y0, x1, y1)) != Box::CLIP_RESULT_UNCHANGED) {
283 if (y0 > y * kBgFillOptimizerWindowSize ||
284 y1 < y * kBgFillOptimizerWindowSize +
285 kBgFillOptimizerWindowSize - 1) {
286 // Need to invalidate the entire line.
287 background_mask_->fillRect(Box(xstart, y, x - 1, y), 0);
288 } else {
289 if (x0 > xstart * kBgFillOptimizerWindowSize) {
290 background_mask_->set(xstart, y, 0);
291 }
292 if (x1 < x * kBgFillOptimizerWindowSize - 1) {
293 background_mask_->set(x - 1, y, 0);
294 }
295 }
296 }
297 filler->fillRect(box.xMin(), box.yMin(), box.xMax(), box.yMax());
298 }
299 xstart = -1;
300 } else {
301 // Continuation, or a new streak.
302 if (xstart < 0) {
303 // New streak.
304 xstart = x;
305 }
306 }
307 }
308 }
309 // Identify the bit-mask rectangle that is entirely covered by the
310 // requested rect. If non-empty, set all nibbles corresponding to it in
311 // the mask, indicating that the area is all-bg.
312 Box inner_filter_box(
313 (x0 + kBgFillOptimizerWindowSize - 1) / kBgFillOptimizerWindowSize,
314 (y0 + kBgFillOptimizerWindowSize - 1) / kBgFillOptimizerWindowSize,
315 (x1 + 1) / kBgFillOptimizerWindowSize - 1,
316 (y1 + 1) / kBgFillOptimizerWindowSize - 1);
317 if (!inner_filter_box.empty()) {
318 background_mask_->fillRect(inner_filter_box, palette_idx);
319 }
320}
