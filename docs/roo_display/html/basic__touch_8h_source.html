11// Base class for touch drivers. Takes care of capping hardware sampling

63template <int max_touch_points>
64TouchResult BasicTouchDevice<max_touch_points>::getTouch(TouchPoint* points,
65 int max_points) {
66 roo_time::Uptime now = roo_time::Uptime::Now();
67 roo_time::Duration dt = now - detection_timestamp_;
68 if (dt < roo_time::Millis(config_.min_sampling_interval_ms)) {
69 // Immediately return the last result without attempting to scan.
70 return pushResult(points, max_points);
71 }
72 TouchPoint readout[max_touch_points];
73 int points_touched = readTouch(readout);
74 DCHECK_GE(points_touched, 0);
75 DCHECK_LE(points_touched, max_touch_points);
76 if (points_touched == 0) {
77 if (dt < roo_time::Millis(config_.touch_intertia_ms)) {
78 // We did not detect touch, but the latest confirmed touch was not long
79 // ago so we report that one anyway, but do not update the
80 // detection_timestamp_ to reflect that we're reporting a stale value.
81 return pushResult(points, max_points);
82 }
83 // Report definitive no touch.
84 detection_timestamp_ = now;
85 points_touched_ = 0;
86 return pushResult(points, max_points);
87 }
88 // Touch has been detected. Need to smooth the values and report it.
89 float alpha = 1 - pow(config_.smoothing_factor, dt.inMicros() / 10000.0);
90 for (int i = 0; i < points_touched; i++) {
91 TouchPoint& p = readout[i];
92 p.vx = 0;
93 p.vy = 0;
94 // Identify a previous touch coordinate, if any.
95 for (int j = 0; j < points_touched_; ++j) {
96 if (touch_points_[j].id == p.id) {
97 TouchPoint& prev = touch_points_[j];
98 // Match! Smooth out the value, and calculate velocity.
99 int16_t x = prev.x * (1 - alpha) + p.x * alpha;
100 int16_t y = prev.y * (1 - alpha) + p.y * alpha;
101 int16_t z = prev.z * (1 - alpha) + p.z * alpha;
102 if (dt > roo_time::Micros(0)) {
103 // Velocity in pixels / s.
104 p.vx = 1000000LL * (x - prev.x) / dt.inMicros();
105 p.vy = 1000000LL * (y - prev.y) / dt.inMicros();
106 }
107 p.x = x;
108 p.y = y;
109 p.z = z;
110 continue;
111 }
112 }
113 }
114 // Copy over and report.
115 detection_timestamp_ = now;
116 points_touched_ = points_touched;
117 std::copy(&readout[0], &readout[points_touched], touch_points_);
118 return pushResult(points, max_points);
119}
