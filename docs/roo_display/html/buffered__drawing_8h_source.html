3// Utilities for buffering pixel writes before they are sent out to underlying
4// device drivers. The buffering has enormous performance impact:
5// * for hardware devices, e.g. SPI-based displays, it allows the driver to
6// optimize commands by noticing patterns like horizontal or vertical streaks,
7// and to amortize per-command overheads in the protocol layer.
8// * for in-memory buffers, eliminating per-pixel virtual calls leads to
9// massive performance improvements as well.
10//
11// Specific utility classes are provided for combinations of:
12// * various write patterns: writing random pixels, subsequent pixels,
13// horizontal lines, vertical lines, rectangles;
14// * using different colors (*Writer) vs a single color (*Filler);
15// * additionally performing pre-clipping (Clipping* variants) or not.
16// The non-clipping variants can be used (and will be slightly faster) when
17// it is known that all drawn primitives fit in the pre-assumed clipping
18// rectangle (e.g., within the hardware-supported bounds).

183 void writeColor(Color color) {
184 color_buffer_[buffer_size_] = color;
185 ++buffer_size_;
186 if (buffer_size_ == kPixelWritingBufferSize) flush();
187 }
188
189 // Writes a number of pixels with the same color.
190 void writeColorN(Color color, uint16_t count) {
191 uint16_t batch = kPixelWritingBufferSize - buffer_size_;
192 if (count < batch) {
193 FillColor(color_buffer_ + buffer_size_, count, color);
194 buffer_size_ += count;
195 return;
196 }
197 FillColor(color_buffer_ + buffer_size_, batch, color);
198 device_.write(color_buffer_, kPixelWritingBufferSize);
199 count -= batch;
200 while (true) {
201 if (count < kPixelWritingBufferSize) {
202 FillColor(color_buffer_, count, color);
203 buffer_size_ = count;
204 return;
205 }
206 FillColor(color_buffer_, kPixelWritingBufferSize, color);
207 device_.write(color_buffer_, kPixelWritingBufferSize);
208 count -= kPixelWritingBufferSize;
209 }
210 }
211
212 // Writes a number of pixels with distinct colors.
213 void writeColors(Color* colors, uint16_t count) {
214 uint16_t batch = kPixelWritingBufferSize - buffer_size_;
215 if (count < batch) {
216 memcpy(color_buffer_ + buffer_size_, colors, count * sizeof(Color));
217 buffer_size_ += count;
218 return;
219 }
220 if (buffer_size_ != 0) {
221 memcpy(color_buffer_ + buffer_size_, colors, batch * sizeof(Color));
222 device_.write(color_buffer_, kPixelWritingBufferSize);
223 count -= batch;
224 colors += batch;
225 }
226 while (true) {
227 if (count < kPixelWritingBufferSize) {
228 memcpy(color_buffer_, colors, count * sizeof(Color));
229 buffer_size_ = count;
230 return;
231 }
232 // For large enough number of pixels, we can skip the buffer entirely.
233 device_.write(colors, kPixelWritingBufferSize);
234 count -= kPixelWritingBufferSize;
235 colors += kPixelWritingBufferSize;
236 }
237 }
238
239 // Returns the number of colors that will stil fit in the buffer
240 // before it needs to be flushed.
241 uint16_t remaining_buffer_space() const {
242 return kPixelWritingBufferSize - buffer_size_;
243 }
244
245 // Gives raw access to the buffer at the current write position.
246 Color* buffer_ptr() { return color_buffer_ + buffer_size_; }
247
248 // Indicates that the subsequent colors have been directly written to the
249 // buffer (via buffer_ptr()), and are considered ready to be written to the
250 // device.
251 void advance_buffer_ptr(uint16_t count) {
252 buffer_size_ += count;
253 if (buffer_size_ >= kPixelWritingBufferSize) {
254 device_.write(color_buffer_, buffer_size_);
255 buffer_size_ = 0;
256 }
257 }
258
259 // Writes no more pixels that can still fit in the memory buffer. If the
260 // buffer gets full, flushes it. Returns the number of colors actually
261 // written.
262 uint16_t writeColorsAligned(Color* colors, uint16_t count) {
263 uint16_t batch = kPixelWritingBufferSize - buffer_size_;
264 if (count < batch) {
265 memcpy(color_buffer_ + buffer_size_, colors, count * sizeof(Color));
266 buffer_size_ += count;
267 return count;
268 }
269 if (buffer_size_ != 0) {
270 memcpy(color_buffer_ + buffer_size_, colors, batch * sizeof(Color));
271 device_.write(color_buffer_, kPixelWritingBufferSize);
272 buffer_size_ = 0;
273 return batch;
274 }
275 device_.write(colors, kPixelWritingBufferSize);
276 return kPixelWritingBufferSize;
277 }

279 ~BufferedColorWriter() { flush(); }
