11// Represents a binary clip mask. Takes a raw data region, where each bit
12// represents a single pixel, and a rectangle that determines the position and
13// size of the mask.
14// The mask is always byte-aligned; i.e., each row always starts on a new byte.
15// It is recommended that the clip mask is aligned within 8-pixels to the
16// underlying device coordinates. It may help avoid needless redraws in case
17// when some low-res framebuf is used under the clip mask.
18class ClipMask {
19 public:
20 ClipMask(const roo::byte* data, Box bounds, bool inverted = false)
21 : data_(data),
22 bounds_(std::move(bounds)),
23 inverted_(inverted),
24 line_width_bytes_((bounds.width() + 7) / 8) {}

29 int>::type = 0>
30 ClipMask(const U* data, Box bounds)
31 : ClipMask((const roo::byte*)data, bounds) {}

33 const roo::byte* data() const { return data_; }
34 const Box& bounds() const { return bounds_; }
35 bool inverted() const { return inverted_; }

37 inline bool isMasked(int16_t x, int16_t y) const {
38 if (!bounds_.contains(x, y)) return inverted_;
39 x -= bounds_.xMin();
40 y -= bounds_.yMin();
41 uint32_t byte_offset = x / 8 + y * line_width_bytes_;
42 uint32_t pixel_index = x % 8;
43 roo::byte b = data_[byte_offset];
44 return (bool)(b & (roo::byte{0x80} >> pixel_index)) ^ inverted_;
45 }
46
47 inline bool isAllMasked(int16_t x, int16_t y, roo::byte mask,
48 uint8_t lines) const {
49 return inverted_ ? isAllUnset(x, y, mask, ~lines)
50 : isAllSet(x, y, mask, lines);
51 }
52
53 inline bool isAllUnmasked(int16_t x, int16_t y, roo::byte mask,
54 uint8_t lines) const {
55 return inverted_ ? isAllSet(x, y, mask, ~lines)
56 : isAllUnset(x, y, mask, lines);
57 }
58
59 void setInverted(bool inverted) { inverted_ = inverted; }
60
61 private:
62 inline bool isAllSet(int16_t x, int16_t y, roo::byte mask,
63 uint8_t lines) const {
64 const roo::byte* ptr = data_ + (x - bounds_.xMin()) / 8 +
65 (y - bounds_.yMin()) * line_width_bytes_;
66 while (lines-- > 0) {
67 if ((*ptr & mask) != mask) return false;
68 ptr += line_width_bytes_;
69 }
70 return true;
71 }
72
73 inline bool isAllUnset(int16_t x, int16_t y, roo::byte mask,
74 uint8_t lines) const {
75 const roo::byte* ptr = data_ + (x - bounds_.xMin()) / 8 +
76 (y - bounds_.yMin()) * line_width_bytes_;
77 while (lines-- > 0) {
78 if ((*ptr & mask) != roo::byte{0}) return false;
79 ptr += line_width_bytes_;
80 }
81 return true;
82 }
83
84 const roo::byte* data_;
85
86 Box bounds_;
87
88 // If true, the meaning of 'set' and 'unset' is inverted.
89 // Normally, 'set' = 'masked out'. If inverted, 'set' = 'unmasked'.
90 bool inverted_;
91
92 uint32_t line_width_bytes_;
93};

186 void fillSingleRect(BlendingMode mode, Color color, int16_t x0, int16_t y0,
187 int16_t x1, int16_t y1) {
188 // Note: we need to flush these every rect, because the rectangles may
189 // be overlapping. (A possible alternative would be to only use
190 // rectfiller).
191 BufferedPixelFiller pfiller(output_, color, mode);
192 BufferedRectFiller rfiller(output_, color, mode);
193 const Box& bounds = clip_mask_.bounds();
194 if (y0 < bounds.yMin()) {
195 if (y1 < bounds.yMin()) {
196 if (!clip_mask_.inverted()) {
197 rfiller.fillRect(x0, y0, x1, y1);
198 }
199 return;
200 }
201 if (!clip_mask_.inverted()) {
202 rfiller.fillRect(x0, y0, x1, bounds.yMin() - 1);
203 }
204 y0 = bounds.yMin();
205 }
206 if (x0 < bounds.xMin()) {
207 if (x1 < bounds.xMin()) {
208 if (!clip_mask_.inverted()) {
209 rfiller.fillRect(x0, y0, x1, y1);
210 }
211 return;
212 }
213 if (!clip_mask_.inverted()) {
214 rfiller.fillRect(x0, y0, bounds.xMin() - 1, y1);
215 }
216 x0 = bounds.xMin();
217 }
218 if (x1 > bounds.xMax()) {
219 if (x0 > bounds.xMax()) {
220 if (!clip_mask_.inverted()) {
221 rfiller.fillRect(x0, y0, x1, y1);
222 }
223 return;
224 }
225 if (!clip_mask_.inverted()) {
226 rfiller.fillRect(bounds.xMax() + 1, y0, x1, y1);
227 }
228 x1 = bounds.xMax();
229 }
230 if (y1 > bounds.yMax()) {
231 if (y0 > bounds.yMax()) {
232 if (!clip_mask_.inverted()) {
233 rfiller.fillRect(x0, y0, x1, y1);
234 }
235 return;
236 }
237 if (!clip_mask_.inverted()) {
238 rfiller.fillRect(x0, bounds.yMax() + 1, x1, y1);
239 }
240 y1 = bounds.yMax();
241 }
242 // Now, [x0, y0, x1, y1] is entirely within bounds.
243 // Iterate in 8x8 blocks, filling (or skipping) the entire block when at all
244 // possible.
245 uint8_t yshift = (y0 - bounds.yMin()) % 8;
246 for (int16_t yc0 = y0; yc0 <= y1;) {
247 int16_t yc1 = yc0 - yshift + 7;
248 if (yc1 > y1) yc1 = y1;
249 uint8_t lines = yc1 - yc0 + 1;
250 uint8_t xshift = (x0 - bounds.xMin()) % 8;
251 for (int16_t xc0 = x0; xc0 <= x1;) {
252 int16_t xc1 = xc0 - xshift + 7;
253 roo::byte mask = roo::byte{0xFF} >> xshift;
254 if (xc1 > x1) {
255 mask &= (roo::byte{0xFF} << (xc1 - x1));
256 xc1 = x1;
257 }
258 if (!clip_mask_.isAllMasked(xc0, yc0, mask, lines)) {
259 if (clip_mask_.isAllUnmasked(xc0, yc0, mask, lines)) {
260 rfiller.fillRect(xc0, yc0, xc1, yc1);
261 } else {
262 // Degenerate to the slow version.
263 for (int16_t y = yc0; y <= yc1; ++y) {
264 for (int16_t x = xc0; x <= xc1; ++x) {
265 if (!clip_mask_.isMasked(x, y)) {
266 pfiller.fillPixel(x, y);
267 }
268 }
269 }
270 }
271 }
272
273 xc0 = xc0 - xshift + 8;
274 xshift = 0;
275 }
276 yc0 = yc0 - yshift + 8;
277 yshift = 0;
278 }
