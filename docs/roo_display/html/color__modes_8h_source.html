11// ColorMode is the following template contract:
12// class T {
13// public:
14// // How many bits are needed to represent a single pixel.
15// static const int8_t bits_per_pixel = ...;
16//
17// // Converts raw value to ARGB8888 color. If bits_per_pixel < 8, the raw
18// // value will be passed in the low-order bits.
19// inline Color toArgbColor(storage_type in) const;
20//
21// // Converts ARGB8888 color to a raw value. If bits_per_pixel < 8, the raw
22// // value is written to the low-order bits.
23// inline storage_type fromArgbColor(Color color) const;
24//
25// // Returns the value that specifies what are possible values of alpha.
26// TransparencyMode transparency() const { return TRANSPARENCY_GRADUAL; }
27// };
28//
29// Additionally, when adding a color mode, you may want to implement a
30// specialized version of RawBlender.

478// Both can be (semi) transparent. Useful for bit masks.
479class Monochrome {

482 : fg_(fg), bg_(bg) {}
483
484 static const int8_t bits_per_pixel = 1;
485

487 return in == 0 ? bg_ : fg_;
488 }
489
490 inline constexpr uint8_t fromArgbColor(Color color) const {
491 return (color == bg_ || color.a() == 0) ? 0 : 1;
492 }
493

496
497 constexpr Color bg() const { return bg_; }
498 void setBg(Color bg) { bg_ = bg; }
499
500 constexpr TransparencyMode transparency() const {
501 return bg().a() == 0xFF ? (fg().a() == 0xFF ? TRANSPARENCY_NONE
502 : fg().a() == 0x00 ? TRANSPARENCY_BINARY
503 : TRANSPARENCY_GRADUAL)
504 : bg().a() == 0x00
505 ? (fg().a() == 0x00 || fg().a() == 0xFF ? TRANSPARENCY_BINARY
506 : TRANSPARENCY_GRADUAL)
507 : TRANSPARENCY_GRADUAL;
508 }

511 return bg().a() != 0xFF || fg().a() != 0xFF;
512 }

516 Color bg_;
517};
