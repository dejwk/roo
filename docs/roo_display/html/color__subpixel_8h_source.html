19// Utility class for color modes whose pixels_per_byte > 1. Allows callers to
20// read and write raw colors at various pixel indexes. This functionality is
21// here, rather than directly in the ColorMode classes, because it is
22// repetitive: identical for any color mode with the same pixels_per_byte.
23//
24// The template contract (implemented by specializations and partial
25// specializations, below) is the following:
26//
27// struct SubPixelColorHelper {
28// // Writes the specified raw_color (which is expected to be stored
29// // in the low bits) to the specified target byte, at the specified
30// // index location (from left-to-right, counting from zero). The
31// // index is in the range [0, pixels_per_byte - 1].
32// void applySubPixelColor(uint8_t raw_color, roo::byte *target, int
33// index);
34//
35// // Reads a raw color from the specified byte, at the specified index
36// // (from left-to-right, counting from zero).
37// uint8_t ReadSubPixelColor(const roo::byte source, int index);
38//
39// // Returns a byte resulting from filling pixels_per_byte pixels
40// // to the same specified raw color.
41// roo::byte RawToFullByte(uint8_t raw_color);
42//
43// // Takes a byte specifying a sequence of raw colors, and converts
44// // them all to ARGB8, storing the resulting pixels_per_byte colors
45// // in the specified result array.
46// inline void ReadSubPixelColorBulk(const ColorMode &mode, roo::byte in,
47// Color *result);
48// };
49
50template <typename ColorMode, ColorPixelOrder pixel_order,
51 int8_t pixels_per_byte = ColorTraits<ColorMode>::pixels_per_byte>
52struct SubPixelColorHelper;

54// Specialization that works for for Monochrome LSB-first.
55template <typename ColorMode>
56struct SubPixelColorHelper<ColorMode, COLOR_PIXEL_ORDER_LSB_FIRST, 8> {
57 void applySubPixelColor(uint8_t raw_color, roo::byte *target, int index) {
58 if (raw_color) {
59 *target |= (roo::byte{0x01} << index);
60 } else {
61 *target &= ~(roo::byte{0x01} << index);
62 }
63 }
64 uint8_t ReadSubPixelColor(const roo::byte source, int index) {
65 return (source & (roo::byte{0x01} << index)) != roo::byte{0};
66 }
67 roo::byte RawToFullByte(uint8_t raw_color) {
68 return raw_color ? roo::byte{0xFF} : roo::byte{0x00};
69 }
70 inline void ReadSubPixelColorBulk(const ColorMode &mode, roo::byte in,
71 Color *result) const {
72 result[0] = mode.toArgbColor((in & roo::byte{0x01}) != roo::byte{0});
73 result[1] = mode.toArgbColor((in & roo::byte{0x02}) != roo::byte{0});

75 result[3] = mode.toArgbColor((in & roo::byte{0x08}) != roo::byte{0});

79 result[7] = mode.toArgbColor((in & roo::byte{0x80}) != roo::byte{0});
80 }
81};
