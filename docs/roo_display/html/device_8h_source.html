103// Base class for display device drivers. In addition to the methods
104// for drawing to the display, also exposes methods to initialize
105// and configure the display.
106//
107// The access pattern supports a notion of a 'transaction' (which
108// may, but does not have to, map to the underlying hardware
109// transaction). The transaction begins with a call to begin(),
110// and ends with a call to end(). Calls to DisplayOutput
111// methods are allowed only within the 'transaction'
112// (i.e., in between begin() and end()). The driver may defer
113// writes, but they must be flushed on end().
114class DisplayDevice : public DisplayOutput {
115 public:
116 virtual ~DisplayDevice() {}
117
118 // Must be called once before any methods on the DisplayOutput interface
119 // are called. Can be called at any time, subsequently, to re-initialize
120 // the display to the default settings. Must be called outside transaction.
121 virtual void init() {}
122
123 // Sets the orientation of the display.
124 void setOrientation(Orientation orientation) {
125 if (orientation_ != orientation) {
126 orientation_ = orientation;
127 orientationUpdated();
128 }
129 }
130
131 // Devices that support orientation change should override this method.
132 virtual void orientationUpdated() {
133 assert(orientation_ == Orientation::Default());
134 }
135
136 // Returns the orientation of the display.
137 Orientation orientation() const { return orientation_; }
138
139 // Returns the width of the display in its native orientation (i.e.,
140 // independent of the orientation actually set on the display). Returns
141 // the same value every time it is called, even if called
142 // before init().
143 int16_t raw_width() const { return raw_width_; }
144
145 // Returns the height of the display in its native orientation (i.e.,
146 // independent of the orientation actually set on the display). Returns
147 // the same value every time it is called, even if called
148 // before init().
149 int16_t raw_height() const { return raw_height_; }
150
151 // Returns the display width, considering the current orientation.
152 int16_t effective_width() const {
153 return orientation().isXYswapped() ? raw_height() : raw_width();
154 }
155
156 // Returns the display height, considering the current orientation.
157 int16_t effective_height() const {
158 return orientation().isXYswapped() ? raw_width() : raw_height();
159 }
160
161 // Devices that support BLENDING_MODE_SOURCE_OVER should ignore this hint.
162 // Devices that do not support alpha-blending, e.g. because the hardware
163 // does not suport reading from the framebuffer, should take this hint
164 // to use the specified color as the assumed background color for any

166 virtual void setBgColorHint(Color bgcolor) {}
