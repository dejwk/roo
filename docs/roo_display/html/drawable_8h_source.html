27// Low-level 'handle' to draw to an underlying device. Passed by the library
28// to Drawable.draw() (see below). Don't create directly.
29//
30// Note that the device uses an absolute coordinate system starting at (0, 0).
31// In order to translate the object's extents to device coordinates in
32// accordance with the surface's possible translation offset, use the following
33// formula:
34//
35// extents.translate(s.dx(), s.dy())
36//
37// Additionally, make sure that the above box is constrained to the clip_box;
38// e.g:
39//
40// Box::Intersect(s.clip_box(), extents.translate(s.dx(), s.dy()))
41//
42// It is often useful to quickly reject objects that are entirely outside the
43// clip box:
44//
45// Box bounds =
46// Box::Intersect(s.clip_box(), extents.translate(s.dx(), s.dy()));
47// if (bounds.empty()) return;
48//
49// In order to compute the clipped bounding box in the object's coordinates,
50// the transformation can be applied in the other direction:
51//
52// Box clipped =
53// Box::Intersect(s.clip_box().translace(-s.dx(), -s.dy()), extents);
54//

188// Interface implemented by any class that represents things (like images in
189// various formats) that can be drawn to an output device (such as a TFT
190// screen, or an off-screen buffer).
191//
192// Implementations must override two methods:
193// * either drawTo or drawInteriorTo (see below), to implement the actual
194// drawing logic;
195// * extents(), returning a bounding box which the drawing fits into,
196// assuming it were drawn at (0, 0).
197class Drawable {
198 public:
199 virtual ~Drawable() {}
200
201 // Returns the bounding box encompassing all pixels that need to be drawn.
202 //
203 // Thie method should is called during an SPI transaction, and should not
204 // block or perform I/O.
205 virtual Box extents() const = 0;
206
207 // Returns the boundaries to be used when aligning this drawable. By default,
208 // equivalent to extents(). Some drawables, notably text labels, may want to
209 // use different bounds for alignment.
210 //
211 // Thie method should is called during an SPI transaction, and should not
212 // block or perform I/O.
213 virtual Box anchorExtents() const { return extents(); }
214
215 // A singleton, representing a 'no-op' drawable with no bounding box.
216 static const Drawable *Empty();
217
218 private:
219 friend void Surface::drawObject(const Drawable &object) const;
220
221 // Draws this object's content, respecting the fill mode. That is, if
222 // s.fill_mode() == FILL_MODE_RECTANGLE, the method must fill the entire
223 // (clipped) extents() rectangle (using s.bgcolor() for transparent parts).
224 //
225 // The default implementation fills the clipped extents() rectangle
226 // with bgcolor, and then calls drawInteriorTo. That causes flicker, so you
227 // should try to override this method if a better implementation is possible.
228 //
229 // The implementation must also respect the other surface's parameters,
230 // particularly the clip_box - i.e. it is not allowed to draw to output
231 // outside of the clip_box.
232 //
233 // The surface's clip-box is always pre-clipped to fit within this drawable's
234 // extents().
235 //
236 virtual void drawTo(const Surface &s) const;
237
238 // Draws this object's content, ignoring s.fill_mode(). See drawTo().
239 //
240 // The implementation most respect the other surface's parameters,
241 // particularly the clip_box - i.e. it is not allowed to draw to output
242 // outside of the clip_box.
243 virtual void drawInteriorTo(const Surface &s) const {}
244};
