177enum FlushMode {
178 // Writes back after every memory write. Slow.
179 FLUSH_MODE_AGGRESSIVE = 0,
180
181 // Writes back after every driver call, i.e. per some number of pixels or
182 // rectangles.
183 FLUSH_MODE_BUFFERED = 1,
184
185 // Only writes back at the end of transaction (in device.end()). In theory,
186 // that sounds like good
187 // idea. And indeed, it seems the fastest. In practice, for some reason it
188 // causes displays to lose synchronization quite a lot.
189 FLUSH_MODE_LAZY = 2,
190};
