132void drawGlyph(const Surface& s, char32_t code, Color color, bool whitespace) {
133 int16_t x = s.dx();
134 int16_t y = s.dy();
135 if (!s.clip_box().intersects(
136 Box(x, y - ascent, x + advance - 1, y - descent + linegap))) {
137 return;
138 }
139 y -= ascent;
140 if (code < 32 || code > 127) return;
141 if (s.fill_mode() == FILL_MODE_VISIBLE) {
142 ClippingBufferedPixelFiller filler(s.out(), color, s.clip_box(),
143 s.blending_mode());
144 for (int8_t i = 0; i < 5; i++) {
145 uint8_t line = pgm_read_byte(font + code * 5 + i);
146 for (int8_t j = 0; j < 8; j++) {
147 if (line & 0x1) filler.fillPixel(x + i, y + j);
148 line >>= 1;
149 }
150 }
151 } else {
152 ClippingBufferedPixelWriter writer(s.out(), s.clip_box(),
153 s.blending_mode());
154 for (int8_t i = 0; i < 5; i++) {
155 uint8_t line = pgm_read_byte(font + code * 5 + i);
156 for (int8_t j = 0; j < 8; j++) {
157 writer.writePixel(x + i, y + j, (line & 0x1) ? color : s.bgcolor());
158 line >>= 1;
159 }
160 if (whitespace) {
161 for (int8_t j = 0; j < 8; j++) {
162 writer.writePixel(x + 5, y + j, s.bgcolor());
163 }
164 }
165 }
166 }

170 const char* utf8_data,
171 uint32_t size,
172 Color color) const {
173 Surface news(s);
174 roo_io::Utf8Decoder decoder(utf8_data, size);
175 char32_t ch;
176 if (!decoder.next(ch)) return;
177 while (true) {
178 char32_t next;
179 bool has_next = decoder.next(next);
180 drawGlyph(news, ch, color, has_next);
181 if (!has_next) return;

183 ch = next;
