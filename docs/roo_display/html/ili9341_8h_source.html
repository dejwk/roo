160 void setOrientation(Orientation orientation) {

169 void setAddrWindow(uint16_t x0, uint16_t y0, uint16_t x1, uint16_t y1)
170 __attribute__((always_inline)) {
171 if (last_x0_ != x0 || last_x1_ != x1) {
172 transport_.sync();
173 writeCommand(CASET);
174 transport_.write16x2_async(x0, x1);
175 last_x0_ = x0;
176 last_x1_ = x1;
177 }
178 if (last_y0_ != y0 || last_y1_ != y1) {
179 transport_.sync();
180 writeCommand(PASET);
181 transport_.write16x2_async(y0, y1);
182 last_y0_ = y0;
183 last_y1_ = y1;
184 }
185 transport_.sync();
186 writeCommand(RAMWR);
187 }
