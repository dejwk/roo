144 void setAddrWindow(uint16_t x0, uint16_t y0, uint16_t x1, uint16_t y1)
145 __attribute__((always_inline)) {
146 if (last_x0_ != x0 || last_x1_ != x1) {
147 transport_.sync();
148 writeCommand(CASET);
149 uint8_t xBin[] = {
150 0, (uint8_t)(x0 >> 8), 0, (uint8_t)(x0 >> 0),
151 0, (uint8_t)(x1 >> 8), 0, (uint8_t)(x1 >> 0),
152 };
153 transport_.writeBytes_async(&xBin[0], 8);
154 last_x0_ = x0;
155 last_x1_ = x1;
156 }
157 if (last_y0_ != y0 || last_y1_ != y1) {
158 transport_.sync();
159 writeCommand(PASET);
160 uint8_t yBin[] = {

163 };
164 transport_.writeBytes_async(&yBin[0], 8);
165 last_y0_ = y0;
