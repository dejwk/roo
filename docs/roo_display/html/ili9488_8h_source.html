20// Stores R, G, B components in high bits of 3 subsequent bytes. Two low bits of
21// each byte are unused.

26 inline constexpr Color toArgbColor(uint32_t in) const {
27 return Color(((in >> 12) & 0xFC) | (in >> 18),
28 ((in >> 4) & 0xFC) | ((in >> 10) & 0x02),
29 ((in << 0) & 0xFC) | ((in >> 2) & 0x02));
30 }
31
32 inline constexpr uint32_t fromArgbColor(Color color) const {
33 return (internal::TruncTo6bit(color.r()) << 18) |
34 (internal::TruncTo6bit(color.g()) << 10) |
35 (internal::TruncTo6bit(color.b()) << 2);
36 }

38 inline uint16_t rawAlphaBlend(uint32_t bg, Color color) const {
39 return fromArgbColor(AlphaBlendOverOpaque(toArgbColor(bg), color));
40 }

43};

207 void ramWrite(uint32_t* data, size_t count) {
208 // Compact the buffer.
209 uint8_t* src = (uint8_t*)data;
210 uint8_t* dest = (uint8_t*)data;
211 size_t byte_count = count * 3;
212 while (count-- > 0) {
213 ++src;
214 *dest++ = *src++;
215 *dest++ = *src++;
216 *dest++ = *src++;
217 }
218 // Write the buffer.
219 transport_.sync();
220 transport_.writeBytes_async((uint8_t*)data, byte_count);
221 }
