193// Run-length-encoded stream with RGB565 color precision and an additional 4-bit
194// alpha channel. The RLE implementation favors fully transparent and fully
195// opaque content.
196template <typename Resource>
197class RleStreamRgb565Alpha4 : public PixelStream {
198 public:
199 RleStreamRgb565Alpha4(const Resource& input)
200 : RleStreamRgb565Alpha4(input.Open()) {}
201
202 RleStreamRgb565Alpha4(StreamType<Resource> input)
203 : input_(std::move(input)),
204 remaining_items_(0),
205 run_rgb_(false),
206 run_value_(0),
207 alpha_buf_(0xFF),
208 alpha_mode_(0) {}
209
210 void Read(Color* buf, uint16_t size) override {
211 while (size-- > 0) {
212 *buf++ = next();
213 }
214 }
215
216 void Skip(uint32_t count) override {
217 while (--count >= 0) next();
218 }
219
220 Color next() {
221 if (remaining_items_ == 0) {
222 // No remaining items; need to decode the next group.
223 uint8_t data = input_.read();
224 // Bit 7 specifies whether all colors in the group have the same RGB
225 // color (possibly with different alpha).
226 run_rgb_ = (data >> 7);
227 // Bits 5-6 determine the type of alpha in the group:
228 // 00: each pixel has a distinct alpha.
229 // 01: pixels have uniform alpha.
230 // 10: all pixels are opaque (alpha = 0xF).
231 // 11: all pixels are transparent (alpha = 0x0).
232 alpha_mode_ = (data & 0x60) >> 5;
233 switch (alpha_mode_) {
234 case 0: {
235 // We have 5 bits left to use, and we only allow odd # of pixels.
236 remaining_items_ =
237 data & 0x10 ? read_varint(data & 0x0F) : data & 0x0F;
238 remaining_items_ <<= 1;
239 break;
240 }
241 case 1: {
242 remaining_items_ = read_varint((data & 0x10) >> 4);
243 alpha_buf_ = (data & 0x0F) * 0x11;
244 break;
245 }
246 case 2: {
247 remaining_items_ =
248 data & 0x10 ? read_varint(data & 0x0F) : data & 0x0F;
249 alpha_buf_ = 0xFF;
250 }
251 case 3: {
252 remaining_items_ =
253 data & 0x10 ? read_varint(data & 0x0F) : data & 0x0F;
254 alpha_buf_ = 0x00;
255 }
256 }
257 if (run_rgb_) {
258 run_value_ = read_color();
259 run_value_.set_a(alpha_buf_);
260 }
261 }
262 --remaining_items_;
263 if (run_rgb_) {
264 if (alpha_mode_ == 0) {
265 if (remaining_items_ & 1) {
266 alpha_buf_ = input_.read();
267 run_value_.set_a((alpha_buf_ >> 4) * 0x11);
268 } else {
269 run_value_.set_a((alpha_buf_ & 0xF) * 0x11);
270 }
271 }
272 return run_value_;
273 } else {
274 Color c;
275 if (alpha_mode_ == 0) {
276 c = read_color();
277 if (remaining_items_ & 1) {
278 alpha_buf_ = input_.read();
279 c.set_a((alpha_buf_ >> 4) * 0x11);
280 } else {
281 c.set_a((alpha_buf_ & 0xF) * 0x11);
282 }
283 } else {
284 c = read_color();
285 c.set_a(alpha_buf_);
286 }
287 return c;
288 }
289 }
290
291 private:
292 Color read_color() {
293 return Rgb565().toArgbColor(roo_io::ReadBeU16(&input_));
294 }
295
296 uint32_t read_varint(uint32_t result) {
297 while (true) {
298 result <= 7;
299 uint8_t datum = input_.read();
300 result |= (datum & 0x7F);
301 if ((datum & 0x80) == 0) return result;
302 }
303 }

346// Run-length-encoded stream of 4-bit nibbles. This RLE implementation favors
347// the extreme values (0x0 and 0xF), and compresses their runs more efficiently
348// than runs of other values. It makes it particularly useful for monochrome or
349// alpha-channel data (e.g. font glyphs) in which the extreme values are much
350// more likely to run.
351template <typename Resource, typename ColorMode>
352class RleStream4bppxBiased<Resource, ColorMode, 4> : public PixelStream {
353 public:
354 RleStream4bppxBiased(const Resource& input, const ColorMode& color_mode)
355 : RleStream4bppxBiased(input.Open(), color_mode) {}
356
357 RleStream4bppxBiased(StreamType<Resource> input, const ColorMode& color_mode)
358 : reader_(std::move(input)),
359 remaining_items_(0),
360 run_(false),
361 run_value_(0),
362 color_mode_(color_mode) {}
363
364 void Read(Color* buf, uint16_t size) override {
365 while (size-- > 0) {
366 *buf++ = next();
367 }
368 }
369
370 void Skip(uint32_t n) override { skip(n); }
371
372 // The algorihm processes nibbles (4-bit values), decoding them as follows:
373 //
374 // * 1xxx (x > 0) : run of x opaque pixels (value = 0xF)
375 // * 0xxx (x > 0) : run of x opaque pixels (value = 0x0)
376 // * 0x0 0x0 cccc: 2 repetitions of color cccc
377 // * 0x0 0xF cccc: 3 repetitions of color cccc
378 // * 0x0 cccc: single color cccc (cccc != 0x0, 0xF)
379 // * 0x8 0x0 <v> cccc: run of (v + 4) repetitions of color cccc
380 // * 0x8 <v> [cccc ...]: (v + 2) arbitrary colors
381 //
382 // where v is a variable-length integer encoded in 3-bit chunks, with
383 // bit 3 indicating whether more chunks follow.
384 //
385 // For example:
386 //
387 // 0x1 -> 0x0 (a single transparent pixel)
388 // 0x3 -> 0x0 0x0 0x0 (run of 3 transparent pixels)
389 // 0x9 -> 0xF (a single opaque pixel)
390 // 0xD -> 0xF 0xF 0xF 0xF 0xF (run of 5 opaque pixels)
391 // 0x0 0x0 0x5 -> 0x5 0x5 (two pixels of value 0x5)
392 // 0x0 0xF 0x5 -> 0x5 0x5 0x5 (three pixels of value 0x5)
393 // 0x8 0x0 0x0 0x5 -> 0x5 0x5 0x5 0x5 (4 pixels of value 0x5)
394 // 0x8 0x0 0x1 0x5 -> 0x5 0x5 0x5 0x5 0x5 (5 pixels of value 0x5)
395 // 0x8 0x1 0x3 0x4 0x5 -> 0x3 0x4 (3 arbitrary pixels: 0x3, 0x4, 0x5)
396 //
397 // 0x8 0xD 0x1 ... -> 13 (11 + 2) arbitrary pixels that follow
398 Color next() {
399 if (remaining_items_ == 0) {
400 // No remaining items; need to decode the next group.
401 uint8_t nibble = (uint8_t)reader_.next();
402 if (nibble == 0x0) {
403 run_ = true;
404 uint8_t operand = (uint8_t)reader_.next();
405 if (operand == 0) {
406 remaining_items_ = 2;
407 run_value_ = color(reader_.next());
408 } else if (operand == 0xF) {
409 remaining_items_ = 3;
410 run_value_ = color(reader_.next());
411 } else {
412 // Singleton value.
413 remaining_items_ = 1;
414 run_value_ = color(operand);
415 }
416 } else if (nibble == 0x8) {
417 int count = read_varint();
418 if (count == 0) {
419 // This actualy means a single zero nibble -> run
420 run_ = true;
421 remaining_items_ = read_varint() + 4;
422 run_value_ = color(reader_.next());
423 } else {
424 // This indicates a list of X+2 arbitrary values
425 run_ = false;
426 remaining_items_ = count + 2;
427 }
428 } else {
429 // Run of opaque or transparent.
430 run_ = true;
431 remaining_items_ = nibble & 0x7;
432 bool transparent = ((nibble & 0x8) == 0);
433 run_value_ = color_mode_.color();
434 if (transparent) {
435 run_value_.set_a(0x0);
436 }
437 }
438 }
439
440 --remaining_items_;
441 if (run_) {
442 if (remaining_items_ == 0) {
443 run_ = false;
444 }
445 return run_value_;
446 } else {
447 return color(reader_.next());
448 }
449 }
450
451 void skip(uint32_t n) {
452 while (n-- > 0) next();
453 }
454
455 TransparencyMode transparency() const { return color_mode_.transparency(); }
456
457 bool ok() const {
458 return reader_.ok();
459 }
460
461 private:
462 inline Color color(uint8_t nibble) { return color_mode_.toArgbColor(nibble); }
463
464 uint32_t read_varint() {
465 uint32_t result = 0;
466 while (true) {
467 uint8_t nibble = reader_.next();
468 result |= (nibble & 7);
469 if ((nibble & 8) == 0) return result;
470 result <<= 3;
471 }
472 }
