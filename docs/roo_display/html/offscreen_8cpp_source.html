26void AddressWindow::setAddress(uint16_t x0, uint16_t y0, uint16_t x1,
27 uint16_t y1, int16_t raw_width,
28 int16_t raw_height, Orientation orientation) {
29 orientation_ = orientation;
30 x0_ = x0;
31 y0_ = y0;
32 x1_ = x1;
33 y1_ = y1;
34 cursor_x_ = x0_;
35 cursor_y_ = y0_;
36 int8_t x_direction = orientation.isLeftToRight() ? 1 : -1;
37 int8_t y_direction = orientation.isTopToBottom() ? 1 : -1;
38
39 if (orientation.isXYswapped()) {
40 offset_ = (x_direction > 0 ? y0_ : raw_width - 1 - y0_) +
41 (y_direction > 0 ? x0_ : raw_height - 1 - x0_) * raw_width;
42 advance_x_ = y_direction * raw_width;
43 advance_y_ = x_direction - y_direction * (x1_ - x0_ + 1) * raw_width;
44 } else {
45 offset_ = (x_direction > 0 ? x0_ : raw_width - 1 - x0_) +
46 (y_direction > 0 ? y0_ : raw_height - 1 - y0_) * raw_width;
47 advance_x_ = x_direction;
48 advance_y_ = y_direction * raw_width - x_direction * (x1_ - x0_ + 1);
49 }
50}
