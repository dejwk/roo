960inline void AddressWindow::advance() {
961 if (cursor_x_ < x1_) {
962 ++cursor_x_;
963 offset_ += advance_x_;
964 } else {
965 cursor_x_ = x0_;
966 ++cursor_y_;
967 offset_ += advance_x_;
968 offset_ += advance_y_;
969 }

972inline void AddressWindow::advance(uint32_t count) {
973 if (count < x1_ - cursor_x_ + 1) {
974 cursor_x_ += count;
975 offset_ += advance_x_ * count;
976 return;
977 }
978 offset_ += advance_x_ * (x1_ - cursor_x_ + 1);
979 offset_ += advance_y_;
980 count -= (x1_ - cursor_x_ + 1);
981 cursor_x_ = x0_;
982 ++cursor_y_;
983 int16_t full_lines = count / width();
984 count = count % width();
985 if (full_lines > 0) {
986 offset_ += full_lines * (advance_x_ * width() + advance_y_);
987 cursor_y_ += full_lines;
988 }
989 cursor_x_ += count;
990 offset_ += advance_x_ * count;
991}
