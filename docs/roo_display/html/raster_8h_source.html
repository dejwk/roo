22// Helper to iterate over pixels read from some underlying resource.
23// Default implementation for color modes with multiple pixels per byte.
24// The resource is interpreted by this reader simply as a non-compressed
25// stream of bytes, containing consecutive pixels. Any semantics of lines and
26// columns of pixels must be handled by the caller.
27template <typename Resource, typename ColorMode, ColorPixelOrder pixel_order,
28 ByteOrder byte_order,
29 int pixels_per_byte = ColorTraits<ColorMode>::pixels_per_byte>
30class RasterPixelStream : public PixelStream {
31 public:
32 RasterPixelStream(StreamType<Resource> stream, const ColorMode& color_mode)
33 : stream_(std::move(stream)),
34 pixel_index_(ColorTraits<ColorMode>::pixels_per_byte),
35 color_mode_(color_mode) {}
36
37 void Read(Color* buf, uint16_t size) override {
38 while (size-- > 0) {
39 *buf++ = next();
40 }
41 }
42
43 void Skip(uint32_t count) override { skip(count); }
44
45 // Advances the iterator to the next pixel in the buffer.
46 Color next() {
47 if (pixel_index_ == ColorTraits<ColorMode>::pixels_per_byte) {
48 pixel_index_ = 0;
49 fetch();
50 }
51 return cache_[pixel_index_++];
52 }
53
54 void skip(uint32_t count) {
55 uint32_t new_pixel_index = count + pixel_index_;
56 if (new_pixel_index <= ColorTraits<ColorMode>::pixels_per_byte) {
57 pixel_index_ = new_pixel_index;
58 return;
59 }
60 uint32_t bytes_to_skip =
61 new_pixel_index / ColorTraits<ColorMode>::pixels_per_byte - 1;
62 if (bytes_to_skip > 0) {
63 stream_.skip(bytes_to_skip);
64 }
65 pixel_index_ = new_pixel_index % ColorTraits<ColorMode>::pixels_per_byte;
66 if (pixel_index_ == 0) {
67 pixel_index_ = ColorTraits<ColorMode>::pixels_per_byte;
68 } else {
69 fetch();
70 }
71 }
72
73 TransparencyMode transparency() const { return color_mode_.transparency(); }
74
75 const ColorMode& color_mode() const { return color_mode_; }
76
77 private:
78 void fetch() {
79 SubPixelColorHelper<ColorMode, pixel_order> subpixel;
80 subpixel.ReadSubPixelColorBulk(color_mode_, stream_.read(), cache_);
81 }
82
83 StreamType<Resource> stream_;
84 uint8_t pixel_index_;
85 const ColorMode& color_mode_;
86 Color cache_[ColorTraits<ColorMode>::pixels_per_byte];
87};
