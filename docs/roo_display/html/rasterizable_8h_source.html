11// Drawable that can provide a color of any point within the extents, given the
12// coordinates. Rasterizables can be used as overlays, backgrounds, and filters.
13//
14// Rasterizable is automatically Drawable and Streamable. All you need to do to
15// implement it is to implement readColors(). That is, if your class has a
16// way of generating a stream of pixels or drawing itself that is more efficient
17// than going pixel-by-pixel, you may want to override those methods. Also, if
18// your rasterizable possibly returns the same color for large areas, you may
19// want to override readColorRect, as it can improve drawing performance when
20// this rasterizable is used e.g. as an overlay or a background.
21class Rasterizable : public virtual Streamable {
22 public:
23 // Read colors corresponding to the specified collection of points, and store
24 // the results in the result array. The caller must ensure that the points are
25 // within this rasterizable's bounds.
26 virtual void readColors(const int16_t* x, const int16_t* y, uint32_t count,
27 Color* result) const = 0;
28
29 // Read colors corresponding to the specified collection of points, and store
30 // the results in the result array. The points may be outside of this
31 // rasterizable's bounds. This function calls readColors after isolating the
32 // points that are within the bounds. For the pixels that are indeed out of
33 // bounds, the function returns out_of_bounds_color.
34 void readColorsMaybeOutOfBounds(
35 const int16_t* x, const int16_t* y, uint32_t count, Color* result,
36 Color out_of_bounds_color = color::Transparent) const;
37
38 // Read colors corresponding to the specified rectangle. Returns true if all
39 // colors are known to be the same. In this case, only the result[0] is
40 // supposed to be read. Otherwise, the result array is filled with colors
41 // corresponding to all the pixels corresponding to the rectangle. The caller
42 // must ensure that the points are within this rasterizable's bounds.
43 virtual bool readColorRect(int16_t xMin, int16_t yMin, int16_t xMax,
44 int16_t yMax, Color* result) const;
45
46 // Default implementation of createStream(), using readColors() to determine
47 // colors of subsequent pixels.
48 std::unique_ptr<PixelStream> createStream() const override;
49
50 // Default implementation of createStream(), using readColors() to determine
51 // colors of subsequent pixels.
52 std::unique_ptr<PixelStream> createStream(const Box& bounds) const override;
53
54 protected:
55 // Default implementation of drawTo(), using readColors() to determine
56 // colors of subsequent pixels.
57 void drawTo(const Surface& s) const override;
58};

65template <typename Getter>
66class SimpleRasterizable : public Rasterizable {
67 public:

73 void readColors(const int16_t* x, const int16_t* y, uint32_t count,
74 Color* result) const override {
75 while (count-- > 0) {
76 *result++ = getter_(*x++, *y++);

117 void readColors(const int16_t* x, const int16_t* y, uint32_t count,

123 extents_.yMin();
