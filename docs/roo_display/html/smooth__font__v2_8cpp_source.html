1// SmoothFontV2 binary format (version 0x0200)
2//
3// All multi-byte integers are big-endian. Offsets are from the start of the
4// font data buffer.
5//
6// Header (variable size; depends on font_metric_bytes and encoding_bytes):
7// u16 version = 0x0200
8// u8 alpha_bits (must be 4)
9// u8 encoding_bytes (1=ASCII, 2=Unicode BMP)
10// u8 font_metric_bytes (1 or 2)
11// u8 offset_bytes (1..3)
12// u8 compression_method (0=none, 1=RLE)
13// u16 glyph_count
14// u16 cmap_entries_count
15// s8/s16 metrics[11] (xMin, yMin, xMax, yMax, ascent, descent,
16// linegap, min_advance, max_advance,
17// max_right_overhang, default_space_width)
18// u8/u16 default_glyph_code
19// u8 kerning_flags (bits 1..0: 0=none, 1=pairs, 2=classes)
20// u16 glyph_metrics_offset (absolute offset to glyph metrics section)
21// u24 glyph_data_offset (absolute offset to glyph data)
22// u8 reserved (must be 0)
23// u8 reserved (must be 0)
24//
25// Cmap section (variable):
26// Repeated cmap_entries_count times (12 bytes each):
27// u16 range_start
28// u16 range_length
29// u16 glyph_id_offset
30// u16 data_entries_count
31// u24 data_offset (to sparse indirection table)
32// u8 format (0=dense, 1=sparse)
33// Sparse indirection tables: for each sparse entry, data_entries_count
34// u16 values listing glyph offsets in the range.
35//
36// Glyph metrics section (offset = glyph_metrics_offset):
37// glyph_count entries, each:
38// s8/s16 xMin (top two bits encode compression flag; see encoder)
39// s8/s16 yMin
40// s8/s16 xMax
41// s8/s16 yMax
42// s8/s16 advance
43// u8/u16/u24 data_offset (offset into glyph data)
44//
45// Kerning section (starts immediately after glyph metrics):
46// Kerning header (type-specific):
47// format 1 (pairs): u16 kerning_pairs_count
48// format 2 (classes):
49// u16 kerning_class_count
50// u16 kerning_source_count
51// u16 kerning_class_entries_count
52// format 0: no data
53// format 1 (pairs): kerning_pairs_count entries (glyph indices):
54// u8/u16 left_glyph_index
55// u8/u16 right_glyph_index
56// u8 weight
57// format 2 (classes, glyph indices):
58// source table (kerning_source_count entries):
59// u8/u16 glyph_index, u8 class_id
60// class table (kerning_class_count entries):
61// u16 entry_offset, u16 entry_count
62// class entries (kerning_class_entries_count entries):
63// u8/u16 dest_glyph_index, u8 weight
64//
65// Glyph data section (offset = glyph_data_offset):
66// Concatenated glyph bitmaps, each compressed or uncompressed per
67// glyph metadata flag.

105 return pgm_read_byte(ptr);

108static int16_t readWord(const roo::byte* PROGMEM ptr) {
109 return (pgm_read_byte(ptr) << 8) | pgm_read_byte(ptr + 1);
110}

112static uint16_t readUWord(const roo::byte* PROGMEM ptr) {
113 return ((uint16_t)pgm_read_byte(ptr) << 8) | pgm_read_byte(ptr + 1);
114}

122class SmoothFontV2::GlyphMetadataReader {
123 public:
124 GlyphMetadataReader(const SmoothFontV2& font, int glyph_index)
125 : font_(font),
126 ptr_(font.glyph_metadata_begin_ +
127 glyph_index * font.glyph_metadata_size_) {}
128
129 GlyphMetrics readMetrics(FontLayout layout, bool& compressed) {
130 int16_t glyphXMin, glyphYMin, glyphXMax, glyphYMax, x_advance;
131 if (font_.font_metric_bytes_ == 1) {
132 uint8_t b = readByte(ptr_ + 0);
133 compressed = ((b >> 7) == ((b >> 6) & 1));
134 glyphXMin = (int8_t)(b ^ ((!compressed) << 6));
135 glyphYMin = readByte(ptr_ + 1);
136 glyphXMax = readByte(ptr_ + 2);
137 glyphYMax = readByte(ptr_ + 3);
138 x_advance = readByte(ptr_ + 4);
139 } else {
140 uint16_t b = readWord(ptr_ + 0);
141 compressed = ((b >> 15) == ((b >> 14) & 1));
142 glyphXMin = (int16_t)(b ^ ((!compressed) << 14));
143 glyphXMin = readWord(ptr_ + 0);
144 glyphYMin = readWord(ptr_ + 2);
145 glyphXMax = readWord(ptr_ + 4);
146 glyphYMax = readWord(ptr_ + 6);
147 x_advance = readWord(ptr_ + 8);
148 }
149 return GlyphMetrics(
150 glyphXMin, glyphYMin, glyphXMax, glyphYMax,
151 layout == FONT_LAYOUT_HORIZONTAL
152 ? x_advance
153 : glyphYMax - glyphYMin + 1 + font_.metrics().linegap());
154 }
155
156 long data_offset() const {
157 int offset_bytes = font_.offset_bytes_;
158 const roo::byte* PROGMEM ptr = ptr_ + 5 * font_.font_metric_bytes_;
159 return offset_bytes == 2
160 ? (pgm_read_byte(ptr) << 8) | pgm_read_byte(ptr + 1)
161 : offset_bytes == 3
162 ? (pgm_read_byte(ptr) << 16) | (pgm_read_byte(ptr + 1) << 8) |
163 pgm_read_byte(ptr + 2)
164 : pgm_read_byte(ptr);
165 }
166
167 private:
168 const SmoothFontV2& font_;
169 const roo::byte* PROGMEM ptr_;
170};

172SmoothFontV2::SmoothFontV2(const roo::byte* font_data PROGMEM)
173 : glyph_count_(0), default_glyph_(0), default_space_width_(0) {
174 roo_io::UnsafeGenericMemoryIterator<const roo::byte PROGMEM*> reader(
175 font_data);
176 uint16_t version = roo_io::ReadBeU16(reader);
177 CHECK(version == 0x0200) << "Unsupported version number: " << version;
178
179 alpha_bits_ = roo_io::ReadU8(reader);
180 CHECK_EQ(alpha_bits_, 4) << "Unsupported alpha encoding: " << alpha_bits_;
181 encoding_bytes_ = roo_io::ReadU8(reader);
182 CHECK_LE(encoding_bytes_, 2)
183 << "Unsupported character encoding: " << encoding_bytes_;
184 font_metric_bytes_ = roo_io::ReadU8(reader);
185 CHECK_LE(font_metric_bytes_, 2)
186 << "Unsupported count of metric bytes: " << encoding_bytes_;
187
188 offset_bytes_ = roo_io::ReadU8(reader);
189 CHECK_LE(offset_bytes_, 3)
190 << "Unsupported count of offset bytes: " << offset_bytes_;
191
192 compression_method_ = roo_io::ReadU8(reader);
193 CHECK_LE(compression_method_, 1)
194 << "Unsupported compression method: " << compression_method_;
195
196 glyph_count_ = roo_io::ReadBeU16(reader);
197 kerning_pairs_count_ = 0;
198 cmap_entries_count_ = roo_io::ReadBeU16(reader);
199
200 FontMetricReader fm_reader(reader, font_metric_bytes_);
201 int xMin = fm_reader.read();
202 int yMin = fm_reader.read();
203 int xMax = fm_reader.read();
204 int yMax = fm_reader.read();
205 int ascent = fm_reader.read();
206 int descent = fm_reader.read();
207 int linegap = fm_reader.read();
208 int min_advance = fm_reader.read();
209 int max_advance = fm_reader.read();
210 int max_right_overhang = fm_reader.read();
211 default_space_width_ = fm_reader.read();
212
213 if (encoding_bytes_ == 1) {
214 default_glyph_ = roo_io::ReadU8(reader);
215 } else {
216 default_glyph_ = roo_io::ReadBeU16(reader);
217 }
218
219 uint8_t kerning_flags = roo_io::ReadU8(reader);
220 kerning_format_ = kerning_flags & 0x03;
221 uint16_t glyph_metrics_offset = roo_io::ReadBeU16(reader);
222 uint32_t glyph_data_offset = ((uint32_t)roo_io::ReadU8(reader) << 16) |
223 ((uint32_t)roo_io::ReadU8(reader) << 8) |
224 (uint32_t)roo_io::ReadU8(reader);
225 roo_io::ReadU8(reader); // Reserved.
226 roo_io::ReadU8(reader); // Reserved.
227 kerning_class_count_ = 0;
228 kerning_source_count_ = 0;
229 kerning_class_entries_count_ = 0;
230
231 font_begin_ = font_data;
232 cmap_entries_begin_ = reader.ptr();
233 const int kCmapEntrySize = 12;
234 if (cmap_entries_count_ > 0) {
235 cmap_entries_ =
236 std::unique_ptr<CmapEntry[]>(new CmapEntry[cmap_entries_count_]);
237 }
238 for (int i = 0; i < cmap_entries_count_; ++i) {
239 const roo::byte* PROGMEM entry = cmap_entries_begin_ + i * kCmapEntrySize;
240 CmapEntry& dst = cmap_entries_[i];
241 dst.range_start = readUWord(entry + 0);
242 uint16_t range_length = readUWord(entry + 2);
243 dst.range_end = (uint32_t)dst.range_start + range_length;
244 dst.glyph_id_offset = readUWord(entry + 4);
245 dst.data_entries_count = readUWord(entry + 6);
246 dst.data_offset = readTriWord(entry + 8);
247 dst.format = readByte(entry + 11);
248 CHECK_LT(dst.data_offset, (1u << 24)) << "Cmap data offset out of range";
249 }
250
251 glyph_metadata_begin_ = font_begin_ + glyph_metrics_offset;
252 glyph_metadata_size_ = (5 * font_metric_bytes_) + offset_bytes_;
253 glyph_kerning_begin_ =
254 glyph_metadata_begin_ + glyph_metadata_size_ * glyph_count_;
255 glyph_data_begin_ = font_begin_ + glyph_data_offset;
256 glyph_index_bytes_ = (glyph_count_ < (1 << 8)) ? 1 : 2;
257
258 kerning_source_table_begin_ = nullptr;
259 kerning_class_table_begin_ = nullptr;
260 kerning_class_entries_begin_ = nullptr;
261 kerning_source_index_bytes_ = 0;
262 kerning_class_entries_count_ = 0;
263 glyph_kerning_size_ = 0;
264
265 if (kerning_format_ == kKerningFormatPairs) {
266 kerning_pairs_count_ = readUWord(glyph_kerning_begin_ + 0);
267 glyph_kerning_begin_ += 2;
268 glyph_kerning_size_ = 2 * glyph_index_bytes_ + 1;
269 // glyph_data_begin_ already set from header.
270 } else if (kerning_format_ == kKerningFormatClasses) {
271 kerning_class_count_ = readUWord(glyph_kerning_begin_ + 0);
272 kerning_source_count_ = readUWord(glyph_kerning_begin_ + 2);
273 kerning_class_entries_count_ = readUWord(glyph_kerning_begin_ + 4);
274 glyph_kerning_begin_ += 6;
275 kerning_source_index_bytes_ = glyph_index_bytes_;
276 kerning_source_table_begin_ = glyph_kerning_begin_;
277 int source_entry_size = kerning_source_index_bytes_ + 1;
278 int source_table_size = kerning_source_count_ * source_entry_size;
279 kerning_class_table_begin_ =
280 kerning_source_table_begin_ + source_table_size;
281 kerning_class_entries_begin_ =
282 kerning_class_table_begin_ + kerning_class_count_ * 4;
283 // kerning_class_entries_count_ read from header.
284 } else {
285 glyph_data_begin_ = glyph_kerning_begin_;
286 }
287
288 Font::init(
289 FontMetrics(ascent, descent, linegap, xMin, yMin, xMax, yMax,
290 max_right_overhang),
291 FontProperties(
292 encoding_bytes_ > 1 ? FontProperties::CHARSET_UNICODE_BMP
293 : FontProperties::CHARSET_ASCII,
294 min_advance == max_advance && kerning_format_ == kKerningFormatNone
295 ? FontProperties::SPACING_MONOSPACE
296 : FontProperties::SPACING_PROPORTIONAL,
297 alpha_bits_ > 1 ? FontProperties::SMOOTHING_GRAYSCALE
298 : FontProperties::SMOOTHING_NONE,
299 kerning_format_ != kKerningFormatNone
300 ? FontProperties::KERNING_PAIRS
301 : FontProperties::KERNING_NONE));
302
303 // Serial.println(String() + "Loaded font with " + glyph_count_ +
304 // " glyphs, size " + (ascent - descent));
305}

307bool is_space(char32_t code) {
308 // http://en.cppreference.com/w/cpp/string/wide/iswspace; see POSIX
309 return code == 0x0020 || code == 0x00A0 ||
310 (code >= 0x0009 && code <= 0x000D) || code == 0x1680 ||
311 code == 0x180E || (code >= 0x2000 && code <= 0x200B) ||
312 code == 0x2028 || code == 0x2029 || code == 0x205F || code == 0x3000 ||
313 code == 0xFEFF;
314}

333void SmoothFontV2::drawBordered(DisplayOutput& output, int16_t x, int16_t y,
334 int16_t bgwidth, const Drawable& glyph,
335 const Box& clip_box, Color bgColor,
336 BlendingMode blending_mode) const {
337 Box outer(x, y - metrics().glyphYMax(), x + bgwidth - 1,
338 y - metrics().glyphYMin());
339
340 // NOTE: bgColor is part of the source, not destination.
341 if (bgColor.a() == 0xFF &&
342 (blending_mode == BLENDING_MODE_SOURCE_OVER ||
343 blending_mode == BLENDING_MODE_SOURCE_OVER_OPAQUE)) {
344 // All souce pixels will be fully opaque.
345 blending_mode = BLENDING_MODE_SOURCE;
346 }
347
348 if (outer.clip(clip_box) == Box::CLIP_RESULT_EMPTY) return;
349 Box inner = glyph.extents().translate(x, y);
350 if (inner.clip(clip_box) == Box::CLIP_RESULT_EMPTY) {
351 output.fillRect(blending_mode, outer, bgColor);
352 return;
353 }
354 if (outer.yMin() < inner.yMin()) {
355 output.fillRect(
356 blending_mode,
357 Box(outer.xMin(), outer.yMin(), outer.xMax(), inner.yMin() - 1),
358 bgColor);
359 }
360 if (outer.xMin() < inner.xMin()) {
361 output.fillRect(
362 blending_mode,
363 Box(outer.xMin(), inner.yMin(), inner.xMin() - 1, inner.yMax()),
364 bgColor);
365 }
366 Surface s(output, x, y, clip_box, false, bgColor, FILL_MODE_RECTANGLE,
367 blending_mode);
368 s.drawObject(glyph);
369 if (outer.xMax() > inner.xMax()) {
370 output.fillRect(
371 blending_mode,
372 Box(inner.xMax() + 1, inner.yMin(), outer.xMax(), inner.yMax()),
373 bgColor);
374 }
375 if (outer.yMax() > inner.yMax()) {
376 output.fillRect(
377 blending_mode,
378 Box(outer.xMin(), inner.yMax() + 1, outer.xMax(), outer.yMax()),
379 bgColor);
380 }
381}

473 void push(char32_t code) {
474 swapped_ = !swapped_;
475 if (is_space(code)) {
476 *mutable_right_metrics() =
477 GlyphMetrics(0, 0, -1, -1, font_->default_space_width_);
478 *mutable_right_data_offset() = 0;
479 *mutable_right_glyph_index() = -1;
480 return;
481 }
482 int glyph_index = font_->findGlyphIndex(code);
483 if (glyph_index == -1) {
484 glyph_index = font_->findGlyphIndex(font_->default_glyph_);
485 }
486 if (glyph_index == -1) {
487 *mutable_right_metrics() =
488 GlyphMetrics(0, 0, -1, -1, font_->default_space_width_);
489 *mutable_right_data_offset() = 0;
490 *mutable_right_glyph_index() = -1;
491 } else {
492 class GlyphMetadataReader glyph_meta(*font_, glyph_index);
493 *mutable_right_metrics() = glyph_meta.readMetrics(
494 FONT_LAYOUT_HORIZONTAL, mutable_right_compressed());
495 *mutable_right_data_offset() = glyph_meta.data_offset();
496 *mutable_right_glyph_index() = glyph_index;
497 }
498 }

501 swapped_ = !swapped_;
502 *mutable_right_glyph_index() = -1;
503 }

509 return swapped_ ? &data_offset_1_ : &data_offset_2_;
510 }
511

523 GlyphMetrics m2_;

525 long data_offset_2_;
526 int glyph_index_1_ = -1;

528 bool compressed_1_;
529 bool compressed_2_;

532GlyphMetrics SmoothFontV2::getHorizontalStringMetrics(const char* utf8_data,
533 uint32_t size) const {
534 roo_io::Utf8Decoder decoder(utf8_data, size);
535 char32_t next_code;
536 if (!decoder.next(next_code)) {
537 // Nothing to draw.
538 return GlyphMetrics(0, 0, -1, -1, 0);
539 }
540 GlyphPairIterator glyphs(this);
541 glyphs.push(next_code);
542 bool has_more;
543 int16_t advance = 0;
544 int16_t yMin = 32767;
545 int16_t yMax = -32768;
546 int16_t xMin = glyphs.right_metrics().lsb();
547 int16_t xMax = xMin;
548 do {
549 char32_t code = next_code;
550 has_more = decoder.next(next_code);
551 int16_t kern;
552 if (has_more) {
553 glyphs.push(next_code);
554 kern = kerning(glyphs.left_glyph_index(), glyphs.right_glyph_index());
555 } else {
556 next_code = 0;
557 glyphs.pushNull();
558 kern = 0;
559 }
560 advance += (glyphs.left_metrics().advance() - kern);
561 const GlyphMetrics& metrics = glyphs.left_metrics();
562 if (yMax < metrics.glyphYMax()) {
563 yMax = metrics.glyphYMax();
564 }
565 if (yMin > metrics.glyphYMin()) {
566 yMin = metrics.glyphYMin();
567 }
568 // NOTE: need to do this at every glyph, because of possibly trailing
569 // spaces.
570 int16_t xm = advance - glyphs.left_metrics().rsb() - 1;
571 if (xm > xMax) {
572 xMax = xm;
573 }
574 } while (has_more);
575 return GlyphMetrics(xMin, yMin, xMax, yMax, advance);
576}

578uint32_t SmoothFontV2::getHorizontalStringGlyphMetrics(
579 const char* utf8_data, uint32_t size, GlyphMetrics* result, uint32_t offset,
580 uint32_t max_count) const {
581 roo_io::Utf8Decoder decoder(utf8_data, size);
582 char32_t next_code;
583 if (!decoder.next(next_code)) {
584 // Nothing to measure.
585 return 0;
586 }
587 uint32_t glyph_idx = 0;
588 uint32_t glyph_count = 0;
589 GlyphPairIterator glyphs(this);
590 glyphs.push(next_code);
591 bool has_more;
592 int16_t advance = 0;
593 do {
594 if (glyph_count >= max_count) return glyph_count;
595 char32_t code = next_code;
596 has_more = decoder.next(next_code);
597 int16_t kern;
598 if (has_more) {
599 glyphs.push(next_code);
600 kern = kerning(glyphs.left_glyph_index(), glyphs.right_glyph_index());
601 } else {
602 next_code = 0;
603 glyphs.pushNull();
604 kern = 0;
605 }
606 if (glyph_idx >= offset) {
607 result[glyph_count++] =
608 GlyphMetrics(glyphs.left_metrics().glyphXMin() + advance,
609 glyphs.left_metrics().glyphYMin(),
610 glyphs.left_metrics().glyphXMax() + advance,
611 glyphs.left_metrics().glyphYMax(),
612 glyphs.left_metrics().advance() + advance);
613 }
614 ++glyph_idx;
615 advance += (glyphs.left_metrics().advance() - kern);
616 } while (has_more);
617 return glyph_count;
618}

620void SmoothFontV2::drawHorizontalString(const Surface& s, const char* utf8_data,
621 uint32_t size, Color color) const {
622 roo_io::Utf8Decoder decoder(utf8_data, size);
623 char32_t next_code;
624 if (!decoder.next(next_code)) {
625 // Nothing to draw.
626 return;
627 }
628 int16_t x = s.dx();
629 int16_t y = s.dy();
630 DisplayOutput& output = s.out();
631
632 GlyphPairIterator glyphs(this);
633 glyphs.push(next_code);
634 int16_t preadvanced = 0;
635 if (glyphs.right_metrics().lsb() < 0) {
636 preadvanced = glyphs.right_metrics().lsb();
637 x += preadvanced;
638 }
639 bool has_more;
640 do {
641 char32_t code = next_code;
642 has_more = decoder.next(next_code);
643 int16_t kern;
644 if (has_more) {
645 glyphs.push(next_code);
646 kern = kerning(glyphs.left_glyph_index(), glyphs.right_glyph_index());
647 } else {
648 next_code = 0;
649 glyphs.pushNull();
650 kern = 0;
651 }
652 if (s.fill_mode() == FILL_MODE_VISIBLE) {
653 // No fill; simply draw and shift.
654 drawGlyphModeVisible(output, x - preadvanced, y, glyphs.left_metrics(),
655 glyphs.left_compressed(), glyphs.left_data(),
656 s.clip_box(), color, s.bgcolor(), s.blending_mode());
657 x += (glyphs.left_metrics().advance() - kern);
658 } else {
659 // General case. We may have two glyphs to worry about, and we may be
660 // pre-advanced. Let's determine our bounding box, taking the
661 // pre-advancement and kerning into account.
662 int16_t advance = glyphs.left_metrics().advance() - kern;
663 int16_t gap = 0;
664 if (has_more) {
665 gap = glyphs.left_metrics().rsb() + glyphs.right_metrics().lsb() - kern;
666 }
667 // Calculate the total width of a rectangle that we will need to fill with
668 // content (glyphs + background).
669 int16_t total_rect_width =
670 glyphs.left_metrics().glyphXMax() + 1 - preadvanced;
671 if (gap < 0) {
672 drawKernedGlyphsModeFill(
673 output, x, y, total_rect_width, glyphs.left_metrics(),
674 glyphs.left_compressed(), glyphs.left_data(), -preadvanced,
675 glyphs.right_metrics(), glyphs.right_compressed(),
676 glyphs.right_data(), advance - preadvanced,
677 Box::Intersect(s.clip_box(), Box(x, y - metrics().glyphYMax(),
678 x + total_rect_width - 1,
679 y - metrics().glyphYMin())),
680 color, s.bgcolor(), s.blending_mode());
681 } else {
682 // Glyphs do not overlap; can draw them one by one.
683 if (has_more) {
684 // Include the interim whitespace gap in the rectangle that we'll fill
685 // (the gap will be filled with background). This way, we draw longer
686 // horizontal strikes which translates to fewer SPI operations.
687 total_rect_width += gap;
688 } else if (glyphs.left_metrics().rsb() > 0) {
689 // After the last character, fill up the right-side bearing.
690 total_rect_width += glyphs.left_metrics().rsb();
691 }
692 drawGlyphModeFill(
693 output, x, y, total_rect_width, glyphs.left_metrics(),
694 glyphs.left_compressed(), glyphs.left_data(), -preadvanced,
695 Box::Intersect(s.clip_box(), Box(x, y - metrics().glyphYMax(),
696 x + total_rect_width - 1,
697 y - metrics().glyphYMin())),
698 color, s.bgcolor(), s.blending_mode());
699 }
700 x += total_rect_width;
701 preadvanced = total_rect_width - (advance - preadvanced);
702 }
703 } while (has_more);
704}

716int SmoothFontV2::findGlyphIndex(char32_t code) const {
717 if (code > 0xFFFF) return -1;
718 uint16_t ucode = (uint16_t)code;
719 for (int i = 0; i < cmap_entries_count_; ++i) {
720 const CmapEntry& entry = cmap_entries_[i];
721 if (ucode < entry.range_start) return -1;
722 if (ucode >= entry.range_end) continue;
723 uint16_t rel = (uint16_t)(ucode - entry.range_start);
724 if (entry.format == 0) {
725 return entry.glyph_id_offset + rel;
726 }
727 if (entry.format == 1) {
728 const roo::byte* PROGMEM data = font_begin_ + entry.data_offset;
729 int lo = 0;
730 int hi = (int)entry.data_entries_count - 1;
731 while (lo <= hi) {
732 int mid = (lo + hi) / 2;
733 uint16_t val = readUWord(data + mid * 2);
734 if (val == rel) return entry.glyph_id_offset + mid;
735 if (rel < val) {
736 hi = mid - 1;
737 } else {
738 lo = mid + 1;
739 }
740 }
741 return -1;
742 }
743 return -1;
744 }
745 return -1;
746}

759int16_t SmoothFontV2::kerning(int left_glyph_index,
760 int right_glyph_index) const {
761 if (left_glyph_index < 0 || right_glyph_index < 0) return 0;
762 if (kerning_format_ == kKerningFormatPairs) {
763 // Pair format: binary search over sorted (left,right) glyph index pairs.
764 int lo = 0;
765 int hi = kerning_pairs_count_ - 1;
766 while (lo <= hi) {
767 int mid = (lo + hi) / 2;
768 const roo::byte* PROGMEM entry =
769 glyph_kerning_begin_ + mid * glyph_kerning_size_;
770 uint16_t left = read_glyph_index(entry, glyph_index_bytes_);
771 uint16_t right =
772 read_glyph_index(entry + glyph_index_bytes_, glyph_index_bytes_);
773 if (left == left_glyph_index && right == right_glyph_index) {
774 return pgm_read_byte(entry + 2 * glyph_index_bytes_);
775 }
776 if (left < left_glyph_index ||
777 (left == left_glyph_index && right < right_glyph_index)) {
778 lo = mid + 1;
779 } else {
780 hi = mid - 1;
781 }
782 }
783 return 0;
784 }
785 if (kerning_format_ == kKerningFormatClasses) {
786 return kerningWithClassFormat(left_glyph_index, right_glyph_index);
787 }
788 return 0;
789}

791int16_t SmoothFontV2::kerningWithClassFormat(int left_glyph_index,
792 int right_glyph_index) const {
793 if (kerning_source_count_ == 0 || kerning_class_count_ == 0) return 0;
794
795 // 1) Find the class assigned to the left glyph using binary search
796 // over the source table (glyph_index -> class_id).
797 int entry_size = kerning_source_index_bytes_ + 1;
798 int lo = 0;
799 int hi = kerning_source_count_ - 1;
800 uint8_t class_id = 0xFF;
801
802 while (lo <= hi) {
803 int mid = (lo + hi) / 2;
804 const roo::byte* PROGMEM entry =
805 kerning_source_table_begin_ + mid * entry_size;
806 uint16_t glyph_index = read_glyph_index(entry, kerning_source_index_bytes_);
807 if (glyph_index == left_glyph_index) {
808 class_id = pgm_read_byte(entry + kerning_source_index_bytes_);
809 break;
810 }
811 if (glyph_index < left_glyph_index) {
812 lo = mid + 1;
813 } else {
814 hi = mid - 1;
815 }
816 }
817
818 if (class_id == 0xFF || class_id >= kerning_class_count_) return 0;
819
820 // 2) Look up the class entry range (offset,count).
821 const roo::byte* PROGMEM cls = kerning_class_table_begin_ + class_id * 4;
822 uint16_t offset = readUWord(cls + 0);
823 uint16_t count = readUWord(cls + 2);
824 if (count == 0) return 0;
825
826 // 3) Binary search within the class entries for the right glyph index.
827 const roo::byte* PROGMEM entries =
828 kerning_class_entries_begin_ + offset * entry_size;
829 lo = 0;
830 hi = count - 1;
831 while (lo <= hi) {
832 int mid = (lo + hi) / 2;
833 const roo::byte* PROGMEM entry = entries + mid * entry_size;
834 uint16_t glyph_index = read_glyph_index(entry, kerning_source_index_bytes_);
835 if (glyph_index == right_glyph_index) {
836 return pgm_read_byte(entry + kerning_source_index_bytes_);
837 }
838 if (glyph_index < right_glyph_index) {
839 lo = mid + 1;
840 } else {
841 hi = mid - 1;
842 }
843 }
844 return 0;
845}
