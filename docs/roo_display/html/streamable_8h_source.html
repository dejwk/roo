284// Restricts the stream to the specified sub-rectangle. More specifically,
285// since streams don't know their rectangle dimension, it:
286// * assumes that the delegate stream has been already skipped to the right
287// starting place, so that the first next() call would return the pixel
288// corresponding to the top-left corner of the sub-rectanble
289// * then, returns the specified number of colored pixels, and then skips
290// by specified amount (to get to the next 'line') in the delegate stream.
291// The caller should guarantee that the sum of these two quantities
292// (width and width_skip) is equal to the pixel width of the image represented
293// by the underlying stream.
294template <typename Delegate>
295class SubRectangleStream : public PixelStream {
296 public:
297 SubRectangleStream(Delegate delegate, uint32_t count, int16_t width,
298 int16_t width_skip)
299 : stream_(std::move(delegate)),
300 x_(0),
301 width_(width),
302 width_skip_(width_skip),
303 remaining_(count),
304 idx_(kPixelWritingBufferSize) {}
305
306 SubRectangleStream(SubRectangleStream&&) = default;
307
308 void Read(Color* buf, uint16_t count) override {
309 do {
310 if (x_ >= width_) {
311 dskip(width_skip_);
312 x_ = 0;
313 }
314 uint16_t n = width_ - x_;
315 if (n > count) n = count;
316 uint16_t read = dnext(buf, n);
317 buf += read;
318 count -= read;
319 x_ += read;
320 } while (count > 0);
321 }

340 uint16_t buffered = kPixelWritingBufferSize - idx_;
341 if (count <= buffered) {
342 idx_ += count;
343 return;

345 count -= buffered;

360 if (idx_ >= kPixelWritingBufferSize) {
361 uint16_t n = kPixelWritingBufferSize;
362 if (n > remaining_) n = remaining_;
363 stream_.Read(buf_, n);
364 idx_ = 0;
365 remaining_ -= n;
366 }
367 if (count > kPixelWritingBufferSize - idx_) {
368 count = kPixelWritingBufferSize - idx_;
369 }
370 memcpy(buf, buf_ + idx_, count * sizeof(Color));
371 idx_ += count;
372 return count;
373 }
