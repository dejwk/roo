101template <typename Spi>
102ConversionResult single_conversion(Spi& spi, uint16_t z_threshold, uint16_t* x,
103 uint16_t* y, uint16_t* z) {
104 // Wait until pressure stops increasing
105 uint16_t z1 = 1;
106 uint16_t z2 = 0;
107 while (z1 > z2) {
108 z2 = z1;
109 z1 = get_raw_touch_z(spi);
110 }
111
112 if (z1 <= z_threshold) {
113 return UNTOUCHED;
114 }
115
116 uint16_t x1, y1, x2, y2;
117 get_raw_touch_xy(spi, &x1, &y1);
118 get_raw_touch_xy(spi, &x2, &y2);
119 if (abs(x1 - x2) > kMaxRawSettlingDistance) return UNSETTLED;
120 if (abs(y1 - y2) > kMaxRawSettlingDistance) return UNSETTLED;
121 uint16_t z3 = get_raw_touch_z(spi);
122 if (z3 <= z_threshold) {
123 return UNSETTLED;
124 }
125
126 // We have a valid touch sample pair. Return an arithmetic average as the
127 // result.
128 *x = (x1 + x2) / 2;
129 *y = (y1 + y2) / 2;
130 *z = z3;
131 return TOUCHED;
132}
