266// Fills 'count' consecutive bits of memory (in MSB order), starting at the
267// given bit offset of the given buffer.
268inline void BitFill(byte* buf, uint32_t offset, int16_t count, bool value) {
269 buf += (offset / 8);
270 offset %= 8;
271 if (value) {
272 if (offset > 0) {
273 if (offset + count < 8) {
274 *buf |= (byte)(((1 << count) - 1) << offset);
275 return;
276 }
277 *buf++ |= (byte)(0xFF << offset);
278 count -= (8 - offset);
279 offset = 0;
280 }
281 memset(buf, 0xFF, count / 8);
282 buf += (count / 8);
283 count %= 8;
284 if (count == 0) return;
285 *buf |= (byte)((1 << count) - 1);
286 } else {
287 if (offset > 0) {
288 if (offset + count < 8) {
289 *buf &= (byte)(~(((1 << count) - 1) << offset));
290 return;
291 }
292 *buf++ &= (byte)(~(0xFF << offset));
293 count -= (8 - offset);
294 offset = 0;
295 }
296 memset(buf, 0x00, count / 8);
297 buf += (count / 8);
298 count %= 8;
299 if (count == 0) return;
300 *buf &= (byte)(~((1 << count) - 1));
301 }
