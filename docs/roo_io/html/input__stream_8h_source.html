31 // Attempts to read up to `count` bytes, but at least one byte, into `result`,
32 // and updates `status()`. Returns the number of bytes read, which must be
33 // greater than zero on success (i.e. when `status()` returns 'kOk'), zero on
34 // end-of-stream (i.e. when `status()` returns 'kEndOfStream'), and possibly
35 // zero on error (i.e. when `status()` returns another value).
36 //
37 // If the status was not 'kOk' prior to the call, leaves it as is and returns
38 // zero. Otherwise, if there are more than zero bytes available before the end
39 // of stream, reads and returns up to `count` bytes, but at least one byte,
40 // and leaves status as `kOk`. Otherwise if there are no more bytes available
41 // before end of stream, returns zero and updates status to `kEndOfStream`.
42 //
43 // In case of error, updates the status, and returns the number of bytes that
44 // have been read before the error was encountered (possibly zero, but might
45 // be greater than zero).
46 //
47 // This method may read fewer than count bytes, even if more are available in
48 // the stream. If that's not what you want, see `readFully()`.
49 virtual size_t read(byte* result, size_t count) = 0;
50
51 // Attempts to read up to `count` bytes, into `result`, as long as it is
52 // possible without blocking indefinitely, and updates `status()`. Returns the
53 // number of bytes read.
54 //
55 // Similar to read(), except that it can return zero on success, in case there
56 // are no more bytes available to read without blocking.
57 //
58 // The implementation has some leeway in deciding what constitutes
59 // unacceptable blocking, as long as it is guaranteed that callers can always
60 // make progress by calling tryRead() repeatedly, and never calling read().
61 virtual size_t tryRead(byte* result, size_t count) {
62 return read(result, count);
63 }

65 // Attempts to read `count` bytes into `result`, and updates `status()`.
66 // Blocks if necessary.
67 //
68 // This method is similar to `read()`, except it never returns fewer bytes
69 // than `count`, unless it reaches end of stream or encounters an error.
70 virtual size_t readFully(byte* buf, size_t count) {
71 size_t read_total = 0;
72 while (count > 0) {
73 int read_now = read(buf, count);
74 if (read_now == 0) break;
75 buf += read_now;
76 read_total += read_now;
77 count -= read_now;
78 }
79 return read_total;
80 }

82 // Skips over the specified count of bytes. Updates the `status()`.
83 //
84 // Conceptually equivalent to `readFully(buf, count)` and ignoring the result.
85 //
86 // If the status was not `kOk` prior to the call, leaves it as is. Otherwise,
87 // if the skip was succesful and the stream is before or exactly at the end,
88 // leaves the status as `kOk`. Otherwise, if the skip moves the stream past
89 // the end, updates the status to `kEndOfStream`.
90 //
91 // In case of error, updates the status accordingly.
92 //
93 virtual void skip(uint64_t count) {
94 byte buf[64];
95 while (count >= 64) {
96 if (status() != kOk) return;
97 count -= read(buf, 64);
98 }
99 while (count > 0) {
100 if (status() != kOk) return;
101 count -= read(buf, count);
102 }
103 }
