44 // Attempts to write up to `count` bytes, to the `buffer`, as long as it is

58 // Attempts to write `count` bytes from the `buffer`. Updates `status()`.
59 // Returns the number of bytes written, which must be `count` on success (i.e.
60 // when `status()` returns `kOk`), and possibly zero on error.
61 //
62 // This method is similar to `write()`, except it never returns fewer bytes
63 // than `count`, unless it encounters an error.
64 //
65 // Since the data may be buffered, successful write does not guarantee that
66 // the data made it all the way through to the sink, until `flush()` is
67 // called.
68 virtual size_t writeFully(const byte* buf, size_t count) {
69 size_t written_total = 0;
70 while (count > 0) {
71 size_t written_now = write(buf, count);
72 if (written_now == 0) break;
73 buf += written_now;
74 written_total += written_now;
75 count -= written_now;
76 }
77 return written_total;
78 }
