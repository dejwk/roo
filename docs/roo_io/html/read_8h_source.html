138// Reads `count` bytes from the input iterator, storing them in the result.
139// Returns the number of bytes successfully read. If the returned value is
140// smaller than `count`, it indicates that the end of stream has been reached,
141// or that an error was encountered. The `status()` of the underlying iterator
142// can be used to determine the cause.
143template <typename InputIterator>
144size_t ReadByteArray(InputIterator& in, byte* result, size_t count) {
145 size_t read_total = 0;
146 while (count > 0) {
147 size_t read_now = in.read(result, count);
148 if (read_now == 0) break;
149 result += read_now;
150 read_total += read_now;
151 count -= read_now;
152 }
153 return read_total;
154}

156// Reads an unsigned 64-bit integer, encoded using variable-length encoding as
157// defined by Google protocol buffers. (Small numbers take little space; numbers
158// up to 127 take 1 byte).
159template <typename InputIterator>
160uint64_t ReadVarU64(InputIterator& in) {
161 uint64_t result = 0;
162 byte read;
163 int shift = 0;
164 do {
165 read = in.read();
166 if (in.status() != kOk) {
167 return 0;
168 }
169 result |= ((uint64_t)(read & byte{0x7F}) << shift);
170 shift += 7;
171 } while ((read & byte{0x80}) != byte{0});
172 return result;
173}

314// Reads a string, represented in portable representation (varint length
315// followed by the character array), as a C string, into the specified buffer
316// `buf` of specified `capacity`, and returns the length of the returned C
317// string.
318//
319// If the string won't fit into the buffer, i.e. if the string's lengh plus 1
320// (to account for the terminating zero) is greater than `capacity`, the
321// returned string gets truncated. As long as `capacity` is greater than zero,
322// the result gets zero-terminated. The return value indicates the actual
323// returned length, not counting the terminal zero.
324//
325// Regardless whether the string gets truncated or not, it is always entirely
326// skipped in the input stream, so that subsequent reads can access the data
327// past the string.
328template <typename InputIterator>
329size_t ReadCString(InputIterator& in, char* buf, size_t capacity = SIZE_MAX) {
330 uint64_t len = ReadVarU64(in);
331 if (in.status() != kOk) return 0;
332 if (len + 1 <= capacity) {
333 // Common case.
334 size_t written = ReadByteArray(in, (byte*)buf, len);
335 buf[written] = 0;
336 return written;
337 }
338 if (capacity > 0) {
339 size_t written = ReadByteArray(in, (byte*)buf, capacity - 1);
340 buf[written] = 0;
341 in.skip(len - written);
342 return written;
343 }
344 in.skip(len);
345 return 0;
346}

348// Reads a string, represented in portable representation (varint length
349// followed by the character array), up to `max_size` length.
350//
351// If the string length exceeds `max_size`, the result gets truncated to
352// `max_size`, but the entire string gets skipped in the input stream.
353template <typename InputIterator>
354std::string ReadString(InputIterator& in, size_t max_size = SIZE_MAX) {
355 uint64_t len = ReadVarU64(in);
356 if (in.status() != kOk) return "";
357 std::string result;
358 if (len <= max_size) {
359 // Common case.
360 result.reserve(len);
361 for (size_t i = 0; i < len; ++i) {
362 char ch = (char)ReadU8(in);
363 if (in.status() != kOk) return result;
364 result.push_back(ch);
365 }
366 return result;
367 }
368 result.reserve(max_size);
369 for (size_t i = 0; i < max_size; ++i) {
370 char ch = (char)ReadU8(in);
371 if (in.status() != kOk) return result;
372 result.push_back(ch);
373 }
374 in.skip(len - max_size);
375 return result;
376}

378// For memory iterators only. Reads a string, represented in portable
379// representation (varint length followed by the character array), as
380// a string_view up to `max_size` length, without copying the underlying data.
381// (The returned string_view is backed by the underlying iterator's memory
382// buffer).
383//
384// If the string length exceeds `max_size`, the result gets truncated to
385// `max_size`, but the entire string gets skipped in the input stream.
386template <typename InputIterator,
387 typename std::enable_if<
388 internal::MemoryIteratorTraits<InputIterator>::is_memory,
389 bool>::type = true>
390roo::string_view ReadStringView(InputIterator& in, size_t max_size = SIZE_MAX) {
391 uint64_t len = ReadVarU64(in);
392 if (in.status() != kOk) return "";
393 typename InputIterator::PtrType start = in.ptr();
394 if (len <= max_size) {
395 // Common case.
396 in.skip(len);
397 return roo::string_view((const char*)start, in.ptr() - start);
398 }
399 in.skip(max_size);
400 roo::string_view result((const char*)start, in.ptr() - start);
401 in.skip(len - max_size);
402 return result;
403}
