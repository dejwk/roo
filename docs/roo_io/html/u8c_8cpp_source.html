17// Returns the length of the next UTF8 character and stores in ch
18// (if it's not null) the corresponding codepoint.
19// It is based on a work by Bjoern Hoehrmann:
20// http://bjoern.hoehrmann.de/utf-8/decoder/dfa
21// but it's faster and has been extended to include C0 80 as the
22// encoding for U+0000

66size_t u8next_(const char *start, const char* end, char32_t &val)
67{
68 size_t len = 0;
69 const unsigned char *s = (const unsigned char *)start;
70 const unsigned char *e = (const unsigned char *)end;
71 if (s == e) return 0;
72 char first = *s;
73
74 val = first;
75 fsm {
76 fsmSTATE(START) {
77 if (*s < 0x80) { len = 1; fsmGOTO(end); }
78 // if (*s == 0xC0) { len = 2; fsmGOTO(null); }
79 if (*s <= 0xC1) { fsmGOTO(invalid); }
80 if (*s <= 0xDF) { val &= 0x1F; len = 2; fsmGOTO(len2_0); }
81 if (*s == 0xE0) { val &= 0x0F; len = 3; fsmGOTO(len3_0); }
82 if (*s <= 0xEC) { val &= 0x0F; len = 3; fsmGOTO(len3_1); }
83 if (*s == 0xED) { val &= 0x0F; len = 3; fsmGOTO(len3_2); }
84 if (*s <= 0xEF) { val &= 0x0F; len = 3; fsmGOTO(len3_1); }
85 if (*s == 0xF0) { val &= 0x07; len = 4; fsmGOTO(len4_0); }
86 if (*s <= 0xF3) { val &= 0x07; len = 4; fsmGOTO(len4_1); }
87 if (*s == 0xF4) { val &= 0x07; len = 4; fsmGOTO(len4_2); }
88 fsmGOTO(invalid);
89 }
90
91 // fsmSTATE(null) { // Modified UTF-8 encodes NULL (U+0000) to C0 80
92 // val = 0;
93 // s++; if (s == e || *s != 0x80) fsmGOTO(invalid);
94 // fsmGOTO(end);
95 // }
96
97 fsmSTATE(len4_0) {
98 s++; if (s == e || *s < 0x90 || 0xBF < *s) fsmGOTO(invalid);
99 fsmGOTO(len4);
100 }
101
102 fsmSTATE(len4_1) {
103 s++; if (s == e || *s < 0x80 || 0xBF < *s) fsmGOTO(invalid);
104 fsmGOTO(len4);
105 }
106
107 fsmSTATE(len4_2) {
108 s++; if (s == e || *s < 0x80 || 0x8F < *s) fsmGOTO(invalid);
109 fsmGOTO(len4);
110 }
111
112 fsmSTATE(len4) {
113 val = (val << 6) | (*s & 0x3F);
114 fsmGOTO(len3_1);
115 }
116
117 fsmSTATE(len3_0) {
118 s++; if (s == e || *s < 0xA0 || 0xBF < *s) fsmGOTO(invalid);
119 fsmGOTO(len3);
120 }
121
122 fsmSTATE(len3_1) {
123 s++; if (s == e || *s < 0x80 || 0xBF < *s) fsmGOTO(invalid);
124 fsmGOTO(len3);
125 }
126
127 fsmSTATE(len3_2) {
128 s++; if (s == e || *s < 0x80 || 0x9F < *s) fsmGOTO(invalid);
129 fsmGOTO(len3);
130 }
131
132 fsmSTATE(len3) {
133 val = (val << 6) | (*s & 0x3F);
134 fsmGOTO(len2_0);
135 }
136
137 fsmSTATE(len2_0) {
138 s++; if (s == e || *s < 0x80 || 0xBF < *s) fsmGOTO(invalid);
139 val = (val << 6) | (*s & 0x3F);
140 fsmGOTO(end);
141 }
142
143 // Try rewinding one byte, but consume at least one.
144 fsmSTATE(invalid) {
145 size_t adv = s - (const unsigned char*)start;
146 len = (adv <= 1 ? 1 : adv);
147 val = 0xFFFD;
148 }
149
150 fsmSTATE(end) { }
151 }
152 return len;
153}

156int u8next_FAST(const char *txt, int *ch)
157{
158 int len = 0;
159 unsigned char *s = (unsigned char *)txt;
160 unsigned char first = *s;
161 int val = 0;
162
163 if (first) {
164 val = first;
165 if (first > 0x7F) {
166 val = *++s & 0x3F;
167 if ((first & 0xF8) == 0xF0) { val |= (first & 0x07)<<6; goto L4;}
168 else if ((first & 0xF0) == 0xE0) { val |= (first & 0x0F)<<6; goto L3;}
169 else { val |= (first & 0x1F)<<6; goto L2;}
170
171 L4: val <<= 6; if (*s) s++; val |= (*s & 0x3F);
172 L3: val <<= 6; if (*s) s++; val |= (*s & 0x3F);
173 L2: ;
174 }
175 len = 1+(s-(unsigned char *)txt);
176 }
177
178 if (ch) *ch = val;
179 return len;
180}
