138// Writes `count` bytes from the `source` to the output iterator. Returns the
139// number of bytes successfully written. If the returned value is smaller than
140// `count`, it indicates that an I/O error has occurred. The `status()` of the
141// underlying iterator can be used to determine the cause.
142template <typename OutputIterator>
143size_t WriteByteArray(OutputIterator& out, const byte* source, size_t count) {
144 size_t written_total = 0;
145 while (count > 0) {
146 size_t written_now = out.write(source, count);
147 if (written_now == 0) break;
148 source += written_now;
149 written_total += written_now;
150 count -= written_now;
151 }
152 return written_total;
153}

155// Writes an unsigned 64-bit integer, encoded using variable-length encoding as
156// defined by Google protocol buffers. (Small numbers take little space; numbers
157// up to 127 take 1 byte).
158template <typename OutputIterator>
159void WriteVarU64(OutputIterator& out, uint64_t data) {
160 byte buffer[10];
161 if (data <= 0x7F) {
162 // Fast-path and special-case for single-byte data.
163 out.write((byte)data);
164 return;
165 }
166
167 size_t size = 0;
168 while (data > 0) {
169 buffer[size++] = ((byte)data & byte{0x7F}) | byte{0x80};
170 data >>= 7;
171 }
172 buffer[size - 1] &= byte{0x7F};
173 out.write(buffer, size);
174}
